48 // Because formatting using c++ streams is stateful, drop down to C is
49 // required Alternatively we could use stringstream, but its performance

51 std::string getFormattedDuration( double duration ) {
52 // Max exponent + 1 is required to represent the whole part
53 // + 1 for decimal point
54 // + 3 for the 3 decimal places
55 // + 1 for null terminator
56 const std::size_t maxDoubleSize = DBL_MAX_10_EXP + 1 + 1 + 3 + 1;
57 char buffer[maxDoubleSize];
58
59 // Save previous errno, to prevent sprintf from overwriting it
60 ErrnoGuard guard;
61#ifdef _MSC_VER
62 size_t printedLength = static_cast<size_t>(
63 sprintf_s( buffer, "%.3f", duration ) );
64#else
65 size_t printedLength = static_cast<size_t>(
66 std::snprintf( buffer, maxDoubleSize, "%.3f", duration ) );
67#endif
68 return std::string( buffer, printedLength );
69 }

82 std::string serializeFilters( std::vector<std::string> const& filters ) {
83 // We add a ' ' separator between each filter
84 size_t serialized_size = filters.size() - 1;
85 for (auto const& filter : filters) {
86 serialized_size += filter.size();
87 }
88
89 std::string serialized;
90 serialized.reserve(serialized_size);
91 bool first = true;
92
93 for (auto const& filter : filters) {
94 if (!first) {
95 serialized.push_back(' ');
96 }
97 first = false;
98 serialized.append(filter);
99 }
100
101 return serialized;
102 }

115 out << "Available reporters:\n";
116 const auto maxNameLen =
117 std::max_element( descriptions.begin(),
118 descriptions.end(),
119 []( ReporterDescription const& lhs,
120 ReporterDescription const& rhs ) {
121 return lhs.name.size() < rhs.name.size();
122 } )
123 ->name.size();
124
125 for ( auto const& desc : descriptions ) {
126 if ( verbosity == Verbosity::Quiet ) {
127 out << TextFlow::Column( desc.name )
128 .indent( 2 )
129 .width( 5 + maxNameLen )
130 << '\n';
131 } else {
132 out << TextFlow::Column( desc.name + ':' )
133 .indent( 2 )
134 .width( 5 + maxNameLen ) +
135 TextFlow::Column( desc.description )
136 .initialIndent( 0 )
137 .indent( 2 )
138 .width( CATCH_CONFIG_CONSOLE_WIDTH - maxNameLen - 8 )
139 << '\n';
140 }
141 }
142 out << '\n' << std::flush;

185 void defaultListTags( std::ostream& out,
186 std::vector<TagInfo> const& tags,
187 bool isFiltered,
188 Verbosity verbosity ) {
189 if (verbosity == Verbosity::Quiet) {
190 for (auto const& tagCount : tags) {
191 out << tagCount.all() << '\n';
192 }
193 return;
194 }
195 if ( isFiltered ) {
196 out << "Tags for matching test cases:\n";
197 } else {
198 out << "All available tags:\n";
199 }
200
201 // minimum whitespace to pad tag counts, possibly overwritten below
202 int maxTagCountLen = 2;
203
204 // determine necessary padding for tag count column
205 if ( ! tags.empty() ) {
206 const auto maxTagCount =
207 std::max_element( tags.begin(),
208 tags.end(),
209 []( auto const& lhs, auto const& rhs ) {
210 return lhs.count < rhs.count;
211 } )
212 ->count;
213
214 // more padding necessary for 3+ digits
215 if (maxTagCount >= 100) {
216 auto numDigits = 1 + std::floor( std::log10( maxTagCount ) );
217 maxTagCountLen = static_cast<int>( numDigits );
218 }
219 }
220
221 for ( auto const& tagCount : tags ) {
222 ReusableStringStream rss;
223 rss << " " << std::setw( maxTagCountLen ) << tagCount.count << " ";
224 auto str = rss.str();
225 auto wrapper = TextFlow::Column( tagCount.all() )
226 .initialIndent( 0 )
227 .indent( str.size() )
228 .width( CATCH_CONFIG_CONSOLE_WIDTH - 10 );
229 out << str << wrapper << '\n';
230 }
231 out << pluralise(tags.size(), "tag"_sr) << "\n\n" << std::flush;
232 }

234 void defaultListTests(std::ostream& out, ColourImpl* streamColour, std::vector<TestCaseHandle> const& tests, bool isFiltered, Verbosity verbosity) {
235 // We special case this to provide the equivalent of old
236 // `--list-test-names-only`, which could then be used by the
237 // `--input-file` option.
238 if (verbosity == Verbosity::Quiet) {
239 listTestNamesOnly(out, tests);
240 return;
241 }
242
243 if (isFiltered) {
244 out << "Matching test cases:\n";
245 } else {
246 out << "All available test cases:\n";
247 }
248
249 for (auto const& test : tests) {
250 auto const& testCaseInfo = test.getTestCaseInfo();
251 Colour::Code colour = testCaseInfo.isHidden()
252 ? Colour::SecondaryText
253 : Colour::None;
254 auto colourGuard = streamColour->guardColour( colour ).engage( out );
255
256 out << TextFlow::Column(testCaseInfo.name).indent(2) << '\n';
257 if (verbosity >= Verbosity::High) {
258 out << TextFlow::Column(Catch::Detail::stringify(testCaseInfo.lineInfo)).indent(4) << '\n';
259 }
260 if (!testCaseInfo.tags.empty() &&
261 verbosity > Verbosity::Quiet) {
262 out << TextFlow::Column(testCaseInfo.tagsAsString()).indent(6) << '\n';
263 }
264 }
265
266 if (isFiltered) {
267 out << pluralise(tests.size(), "matching test case"_sr);
268 } else {
269 out << pluralise(tests.size(), "test case"_sr);
270 }
271 out << "\n\n" << std::flush;
272 }

275 class SummaryColumn {
276 public:
277 SummaryColumn( std::string suffix, Colour::Code colour ):
278 m_suffix( CATCH_MOVE( suffix ) ), m_colour( colour ) {}
279
280 SummaryColumn&& addRow( std::uint64_t count ) && {
281 std::string row = std::to_string(count);
282 auto const new_width = std::max( m_width, row.size() );
283 if ( new_width > m_width ) {
284 for ( auto& oldRow : m_rows ) {
285 oldRow.insert( 0, new_width - m_width, ' ' );
286 }
287 } else {
288 row.insert( 0, m_width - row.size(), ' ' );
289 }
290 m_width = new_width;
291 m_rows.push_back( row );
292 return std::move( *this );
293 }
294
295 std::string const& getSuffix() const { return m_suffix; }
296 Colour::Code getColour() const { return m_colour; }
297 std::string const& getRow( std::size_t index ) const {
298 return m_rows[index];
299 }
300
301 private:
302 std::string m_suffix;
303 Colour::Code m_colour;
304 std::size_t m_width = 0;
305 std::vector<std::string> m_rows;
306 };

308 void printSummaryRow( std::ostream& stream,
309 ColourImpl& colour,
310 StringRef label,
311 std::vector<SummaryColumn> const& cols,
312 std::size_t row ) {
313 for ( auto const& col : cols ) {
314 auto const& value = col.getRow( row );
315 auto const& suffix = col.getSuffix();
316 if ( suffix.empty() ) {
317 stream << label << ": ";
318 if ( value != "0" ) {
319 stream << value;
320 } else {
321 stream << colour.guardColour( Colour::Warning )
322 << "- none -";
323 }
324 } else if ( value != "0" ) {
325 stream << colour.guardColour( Colour::LightGrey ) << " | "
326 << colour.guardColour( col.getColour() ) << value
327 << ' ' << suffix;
328 }
329 }
330 stream << '\n';
331 }

334 void printTestRunTotals( std::ostream& stream,
335 ColourImpl& streamColour,
336 Totals const& totals ) {
337 if ( totals.testCases.total() == 0 ) {
338 stream << streamColour.guardColour( Colour::Warning )
339 << "No tests ran\n";
340 return;
341 }
342
343 if ( totals.assertions.total() > 0 && totals.testCases.allPassed() ) {
344 stream << streamColour.guardColour( Colour::ResultSuccess )
345 << "All tests passed";
346 stream << " ("
347 << pluralise( totals.assertions.passed, "assertion"_sr )
348 << " in "
349 << pluralise( totals.testCases.passed, "test case"_sr )
350 << ')' << '\n';
351 return;
352 }
353
354 std::vector<SummaryColumn> columns;
355 // Don't include "skipped assertions" in total count
356 const auto totalAssertionCount =
357 totals.assertions.total() - totals.assertions.skipped;
358 columns.push_back( SummaryColumn( "", Colour::None )
359 .addRow( totals.testCases.total() )
360 .addRow( totalAssertionCount ) );
361 columns.push_back( SummaryColumn( "passed", Colour::Success )
362 .addRow( totals.testCases.passed )
363 .addRow( totals.assertions.passed ) );
364 columns.push_back( SummaryColumn( "failed", Colour::ResultError )
365 .addRow( totals.testCases.failed )
366 .addRow( totals.assertions.failed ) );
367 columns.push_back( SummaryColumn( "skipped", Colour::Skip )
368 .addRow( totals.testCases.skipped )
369 // Don't print "skipped assertions"
370 .addRow( 0 ) );
371 columns.push_back(
372 SummaryColumn( "failed as expected", Colour::ResultExpectedFailure )
373 .addRow( totals.testCases.failedButOk )
374 .addRow( totals.assertions.failedButOk ) );
375 printSummaryRow( stream, streamColour, "test cases"_sr, columns, 0 );
376 printSummaryRow( stream, streamColour, "assertions"_sr, columns, 1 );
377 }
