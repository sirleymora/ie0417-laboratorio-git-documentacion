29 void listTags(IEventListener& reporter, IConfig const& config) {
30 auto const& testSpec = config.testSpec();
31 std::vector<TestCaseHandle> matchedTestCases = filterTests(getAllTestCasesSorted(config), testSpec, config);
32
33 std::map<StringRef, TagInfo, Detail::CaseInsensitiveLess> tagCounts;
34 for (auto const& testCase : matchedTestCases) {
35 for (auto const& tagName : testCase.getTestCaseInfo().tags) {
36 auto it = tagCounts.find(tagName.original);
37 if (it == tagCounts.end())
38 it = tagCounts.insert(std::make_pair(tagName.original, TagInfo())).first;
39 it->second.add(tagName.original);
40 }
41 }
42
43 std::vector<TagInfo> infos; infos.reserve(tagCounts.size());
44 for (auto& tagc : tagCounts) {
45 infos.push_back(CATCH_MOVE(tagc.second));
46 }
47
48 reporter.listTags(infos);
49 }

83 std::string TagInfo::all() const {
84 // 2 per tag for brackets '[' and ']'
85 size_t size = spellings.size() * 2;
86 for (auto const& spelling : spellings) {
87 size += spelling.size();
88 }
89
90 std::string out; out.reserve(size);
91 for (auto const& spelling : spellings) {
92 out += '[';
93 out += spelling;
94 out += ']';
95 }
96 return out;
97 }
