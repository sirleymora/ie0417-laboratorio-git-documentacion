82 // Returns a substring of [start, start + length).
83 // If start + length > size(), then the substring is [start, size()).
84 // If start > size(), then the substring is empty.
85 constexpr StringRef substr(size_type start, size_type length) const noexcept {
86 if (start < m_size) {
87 const auto shortened_size = m_size - start;
88 return StringRef(m_start + start, (shortened_size < length) ? shortened_size : length);
89 } else {
90 return StringRef();
