53 ITracker* ITracker::findChild( NameAndLocationRef const& nameAndLocation ) {
54 auto it = std::find_if(
55 m_children.begin(),
56 m_children.end(),
57 [&nameAndLocation]( ITrackerPtr const& tracker ) {
58 auto const& tnameAndLoc = tracker->nameAndLocation();
59 if ( tnameAndLoc.location.line !=
60 nameAndLocation.location.line ) {
61 return false;
62 }
63 return tnameAndLoc == nameAndLocation;
64 } );
65 return ( it != m_children.end() ) ? it->get() : nullptr;
66 }

125 void TrackerBase::close() {
126
127 // Close any still open children (e.g. generators)
128 while( &m_ctx.currentTracker() != this )
129 m_ctx.currentTracker().close();
130
131 switch( m_runState ) {
132 case NeedsAnotherRun:
133 break;
134
135 case Executing:
136 m_runState = CompletedSuccessfully;
137 break;
138 case ExecutingChildren:
139 if( std::all_of(m_children.begin(), m_children.end(), [](ITrackerPtr const& t){ return t->isComplete(); }) )
140 m_runState = CompletedSuccessfully;
141 break;
142
143 case NotStarted:
144 case CompletedSuccessfully:
145 case Failed:
146 CATCH_INTERNAL_ERROR( "Illogical state: " << m_runState );
147
148 default:
149 CATCH_INTERNAL_ERROR( "Unknown state: " << m_runState );
150 }
151 moveToParent();
152 m_ctx.completeCycle();
153 }
154 void TrackerBase::fail() {
155 m_runState = Failed;
156 if( m_parent )
157 m_parent->markAsNeedingAnotherRun();
158 moveToParent();
159 m_ctx.completeCycle();
160 }

170 bool SectionTracker::isFilteredImpl() const {
171 // TBD: This is currently _very_ similar to the block in `isComplete`.
172 // Is this neccessarily that way, or just accident of current semantics?
173 const size_t filterIndex =
174 m_newStyleFilters ? m_allTrackerDepth : m_sectionOnlyDepth;
175
176 if ( filterIndex < m_filterRef->size() ) {
177 // 1) New style filter must explicitly target section
178 if ( m_newStyleFilters && ( *m_filterRef )[filterIndex].type !=
179 PathFilter::For::Section ) {
180 return true;
181 }
182 // 2) Both style filters must match the trimmed name exactly
183 if ( m_trimmed_name !=
184 StringRef( ( *m_filterRef )[filterIndex].filter ) ) {
185 return true;
186 }
187 }
188
189 return false;
190 }

201 bool SectionTracker::isComplete() const {
202 // If there are active filters AND we do not pass them,
203 // the section is always "completed"
204 const size_t filterIndex =
205 m_newStyleFilters ? m_allTrackerDepth : m_sectionOnlyDepth;
206 if ( filterIndex < m_filterRef->size() ) {
207 // There is active filter, check it
208 // 1) New style filter must explicitly target section
209 if ( m_newStyleFilters && ( *m_filterRef )[filterIndex].type !=
210 PathFilter::For::Section ) {
211 return true;
212 }
213 // 2) Both style filters must match the trimmed name exactly
214 if ( m_trimmed_name !=
215 StringRef( ( *m_filterRef )[filterIndex].filter ) ) {
216 return true;
217 }
218 }
219
220 // Otherwise we delegate to the generic processing
221 return TrackerBase::isComplete();
222 }

226 SectionTracker& SectionTracker::acquire( TrackerContext& ctx, NameAndLocationRef const& nameAndLocation ) {
227 SectionTracker* tracker;
228
229 ITracker& currentTracker = ctx.currentTracker();
230 if ( ITracker* childTracker =
231 currentTracker.findChild( nameAndLocation ) ) {
232 assert( childTracker );
233 assert( childTracker->isSectionTracker() );
234 tracker = static_cast<SectionTracker*>( childTracker );
235 } else {
236 auto newTracker = Catch::Detail::make_unique<SectionTracker>(
237 NameAndLocation{ static_cast<std::string>(nameAndLocation.name),
238 nameAndLocation.location },
239 ctx,
240 &currentTracker );
241 tracker = newTracker.get();
242 currentTracker.addChild( CATCH_MOVE( newTracker ) );
243 }
244
245 if ( !ctx.completedCycle() ) {
246 tracker->tryOpen();
247 }
248
249 return *tracker;
250 }
