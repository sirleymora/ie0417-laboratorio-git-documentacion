28 auto const setWarning = [&]( std::string const& warning ) {
29 if ( warning == "NoAssertions" ) {
30 config.warnings = static_cast<WarnAbout::What>(config.warnings | WarnAbout::NoAssertions);
31 return ParserResult::ok( ParseResultType::Matched );
32 } else if ( warning == "UnmatchedTestSpec" ) {
33 config.warnings = static_cast<WarnAbout::What>(config.warnings | WarnAbout::UnmatchedTestSpec);
34 return ParserResult::ok( ParseResultType::Matched );
35 } else if ( warning == "InfiniteGenerators" ) {
36 config.warnings = static_cast<WarnAbout::What>(config.warnings | WarnAbout::InfiniteGenerator);
37 return ParserResult::ok( ParseResultType::Matched );
38 }
39
40 return ParserResult ::runtimeError(
41 "Unrecognised warning option: '" + warning + '\'' );
42 };
43 auto const loadTestNamesFromFile = [&]( std::string const& filename ) {
44 std::ifstream f( filename.c_str() );
45 if( !f.is_open() )
46 return ParserResult::runtimeError( "Unable to load input file: '" + filename + '\'' );
47
48 std::string line;
49 while( std::getline( f, line ) ) {
50 line = trim(line);
51 if( !line.empty() && !startsWith( line, '#' ) ) {
52 if( !startsWith( line, '"' ) )
53 line = '"' + CATCH_MOVE(line) + '"';
54 config.testsOrTags.push_back( line );
55 config.testsOrTags.emplace_back( "," );
56 }
57 }
58 //Remove comma in the end
59 if(!config.testsOrTags.empty())
60 config.testsOrTags.erase( config.testsOrTags.end()-1 );
61
62 return ParserResult::ok( ParseResultType::Matched );
63 };
64 auto const setTestOrder = [&]( std::string const& order ) {
65 if( startsWith( "declared", order ) )
66 config.runOrder = TestRunOrder::Declared;
67 else if( startsWith( "lexical", order ) )
68 config.runOrder = TestRunOrder::LexicographicallySorted;
69 else if( startsWith( "random", order ) )
70 config.runOrder = TestRunOrder::Randomized;
71 else
72 return ParserResult::runtimeError( "Unrecognised ordering: '" + order + '\'' );
73 return ParserResult::ok( ParseResultType::Matched );
74 };
75 auto const setRngSeed = [&]( std::string const& seed ) {
76 if( seed == "time" ) {
77 config.rngSeed = generateRandomSeed(GenerateFrom::Time);
78 config.rngSeedWasFixed = false;
79 return ParserResult::ok(ParseResultType::Matched);
80 } else if (seed == "random-device") {
81 config.rngSeed = generateRandomSeed(GenerateFrom::RandomDevice);
82 config.rngSeedWasFixed = false;
83 return ParserResult::ok(ParseResultType::Matched);
84 }
85
86 // TODO: ideally we should be parsing uint32_t directly
87 // fix this later when we add new parse overload
88 auto parsedSeed = parseUInt( seed, 0 );
89 if ( !parsedSeed ) {
90 return ParserResult::runtimeError( "Could not parse '" + seed + "' as seed" );
91 }
92 config.rngSeed = *parsedSeed;
93 config.rngSeedWasFixed = true;
94 return ParserResult::ok( ParseResultType::Matched );
95 };
96 auto const setDefaultColourMode = [&]( std::string const& colourMode ) {
97 Optional<ColourMode> maybeMode = Catch::Detail::stringToColourMode(toLower( colourMode ));
98 if ( !maybeMode ) {
99 return ParserResult::runtimeError(
100 "colour mode must be one of: default, ansi, win32, "
101 "or none. '" +
102 colourMode + "' is not recognised" );
103 }
104 auto mode = *maybeMode;
105 if ( !isColourImplAvailable( mode ) ) {
106 return ParserResult::runtimeError(
107 "colour mode '" + colourMode +
108 "' is not supported in this binary" );
109 }
110 config.defaultColourMode = mode;
111 return ParserResult::ok( ParseResultType::Matched );
112 };
113 auto const setWaitForKeypress = [&]( std::string const& keypress ) {
114 auto keypressLc = toLower( keypress );
115 if (keypressLc == "never")
116 config.waitForKeypress = WaitForKeypress::Never;
117 else if( keypressLc == "start" )
118 config.waitForKeypress = WaitForKeypress::BeforeStart;
119 else if( keypressLc == "exit" )
120 config.waitForKeypress = WaitForKeypress::BeforeExit;
121 else if( keypressLc == "both" )
122 config.waitForKeypress = WaitForKeypress::BeforeStartAndExit;
123 else
124 return ParserResult::runtimeError( "keypress argument must be one of: never, start, exit or both. '" + keypress + "' not recognised" );
125 return ParserResult::ok( ParseResultType::Matched );
126 };
127 auto const setVerbosity = [&]( std::string const& verbosity ) {
128 auto lcVerbosity = toLower( verbosity );
129 if( lcVerbosity == "quiet" )
130 config.verbosity = Verbosity::Quiet;
131 else if( lcVerbosity == "normal" )
132 config.verbosity = Verbosity::Normal;
133 else if( lcVerbosity == "high" )
134 config.verbosity = Verbosity::High;
135 else
136 return ParserResult::runtimeError( "Unrecognised verbosity, '" + verbosity + '\'' );
137 return ParserResult::ok( ParseResultType::Matched );
138 };
139 auto const setReporter = [&]( std::string const& userReporterSpec ) {
140 if ( userReporterSpec.empty() ) {
141 return ParserResult::runtimeError( "Received empty reporter spec." );
142 }
143
144 Optional<ReporterSpec> parsed =
145 parseReporterSpec( userReporterSpec );
146 if ( !parsed ) {
147 return ParserResult::runtimeError(
148 "Could not parse reporter spec '" + userReporterSpec +
149 "'" );
150 }
151
152 auto const& reporterSpec = *parsed;
153
154 auto const& factories =
155 getRegistryHub().getReporterRegistry().getFactories();
156 auto result = factories.find( reporterSpec.name() );
157
158 if ( result == factories.end() ) {
159 return ParserResult::runtimeError(
160 "Unrecognized reporter, '" + reporterSpec.name() +
161 "'. Check available with --list-reporters" );
162 }
163
164
165 const bool hadOutputFile = reporterSpec.outputFile().some();
166 config.reporterSpecifications.push_back( CATCH_MOVE( *parsed ) );
167 // It would be enough to check this only once at the very end, but
168 // there is not a place where we could call this check, so do it
169 // every time it could fail. For valid inputs, this is still called
170 // at most once.
171 if (!hadOutputFile) {
172 int n_reporters_without_file = 0;
173 for (auto const& spec : config.reporterSpecifications) {
174 if (spec.outputFile().none()) {
175 n_reporters_without_file++;
176 }
177 }
178 if (n_reporters_without_file > 1) {
179 return ParserResult::runtimeError( "Only one reporter may have unspecified output file." );
180 }
181 }
182
183 return ParserResult::ok( ParseResultType::Matched );
184 };
185 auto const setShardCount = [&]( std::string const& shardCount ) {
186 auto parsedCount = parseUInt( shardCount );
187 if ( !parsedCount ) {
188 return ParserResult::runtimeError(
189 "Could not parse '" + shardCount + "' as shard count" );
190 }
191 if ( *parsedCount == 0 ) {
192 return ParserResult::runtimeError(
193 "Shard count must be positive" );
194 }
195 config.shardCount = *parsedCount;
196 return ParserResult::ok( ParseResultType::Matched );
197 };
198 auto const setBenchmarkSamples = [&]( std::string const& samples ) {
199 auto parsedSamples = parseUInt( samples );
200 if ( !parsedSamples ) {
201 return ParserResult::runtimeError(
202 "Could not parse '" + samples + "' as benchmark samples" );
203 }
204 if ( *parsedSamples == 0 ) {
205 return ParserResult::runtimeError(
206 "Benchmark samples must be greater than 0" );
207 }
208 config.benchmarkSamples = *parsedSamples;
209 return ParserResult::ok( ParseResultType::Matched );
210 };
211
212 auto const setShardIndex = [&](std::string const& shardIndex) {
213 auto parsedIndex = parseUInt( shardIndex );
214 if ( !parsedIndex ) {
215 return ParserResult::runtimeError(
216 "Could not parse '" + shardIndex + "' as shard index" );
217 }
218 config.shardIndex = *parsedIndex;
219 return ParserResult::ok( ParseResultType::Matched );
220 };
221
222 auto const setSectionFilter = [&]( std::string const& sectionFilter ) {
223 config.pathFilters.emplace_back( PathFilter::For::Section, trim(sectionFilter) );
224 return ParserResult::ok( ParseResultType::Matched );
225 };
226 auto const setGeneratorFilter = [&]( std::string const& generatorFilter ) {
227 if (generatorFilter != "*") {
228 // TODO: avoid re-parsing the index?
229 auto parsedIndex = parseUInt( generatorFilter );
230 if ( !parsedIndex ) {
231 return ParserResult::runtimeError( "Could not parse '" +
232 generatorFilter +
233 "' as generator index" );
234 }
235 }
236 config.useNewPathFilteringBehaviour = true;
237 config.pathFilters.emplace_back( PathFilter::For::Generator, trim(generatorFilter) );
238 return ParserResult::ok( ParseResultType::Matched );
239 };
240 // Copy-capturing other `setFoo` functions enables calling them later,
241 // as the config ref remains valid, but the local lambda vars won't.
242 auto const setPathFilter = [=, &config]( std::string const& pathFilter ) {
243 config.useNewPathFilteringBehaviour = true;
244 if ( pathFilter.size() < 3 ) {
245 return ParserResult::runtimeError(
246 "Path filter '" + pathFilter + "' is too short" );
247 }
248 if ( startsWith( pathFilter, "g:" ) ) {
249 return setGeneratorFilter( pathFilter.substr( 2 ) );
250 }
251 if ( startsWith( pathFilter, "c:" ) ) {
252 return setSectionFilter( pathFilter.substr( 2 ) );
253 }
254 return ParserResult::runtimeError( "Path filter '" + pathFilter +
255 "' has unknown type prefix" );
256 };
