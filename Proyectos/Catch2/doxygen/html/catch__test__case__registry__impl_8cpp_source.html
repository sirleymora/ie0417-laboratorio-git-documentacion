28 static void enforceNoDuplicateTestCases(
29 std::vector<TestCaseHandle> const& tests ) {
30 auto testInfoCmp = []( TestCaseInfo const* lhs,
31 TestCaseInfo const* rhs ) {
32 return *lhs < *rhs;
33 };
34 std::set<TestCaseInfo const*, decltype( testInfoCmp )&> seenTests(
35 testInfoCmp );
36 for ( auto const& test : tests ) {
37 const auto infoPtr = &test.getTestCaseInfo();
38 const auto prev = seenTests.insert( infoPtr );
39 CATCH_ENFORCE( prev.second,
40 "error: test case \""
41 << infoPtr->name << "\", with tags \""
42 << infoPtr->tagsAsString()
43 << "\" already defined.\n"
44 << "\tFirst seen at "
45 << ( *prev.first )->lineInfo << "\n"
46 << "\tRedefined at " << infoPtr->lineInfo );
47 }
48 }

59 std::vector<TestCaseHandle> sortTests( IConfig const& config, std::vector<TestCaseHandle> const& unsortedTestCases ) {
60 switch (config.runOrder()) {
61 case TestRunOrder::Declared:
62 return unsortedTestCases;
63
64 case TestRunOrder::LexicographicallySorted: {
65 std::vector<TestCaseHandle> sorted = unsortedTestCases;
66 std::sort(
67 sorted.begin(),
68 sorted.end(),
69 []( TestCaseHandle const& lhs, TestCaseHandle const& rhs ) {
70 return lhs.getTestCaseInfo() < rhs.getTestCaseInfo();
71 }
72 );
73 return sorted;
74 }
75 case TestRunOrder::Randomized: {
76 using TestWithHash = std::pair<TestCaseInfoHasher::hash_t, TestCaseHandle>;
77
78 TestCaseInfoHasher h{ config.rngSeed() };
79 std::vector<TestWithHash> indexed_tests;
80 indexed_tests.reserve(unsortedTestCases.size());
81
82 for (auto const& handle : unsortedTestCases) {
83 indexed_tests.emplace_back(h(handle.getTestCaseInfo()), handle);
84 }
85
86 std::sort( indexed_tests.begin(),
87 indexed_tests.end(),
88 []( TestWithHash const& lhs, TestWithHash const& rhs ) {
89 if ( lhs.first == rhs.first ) {
90 return lhs.second.getTestCaseInfo() <
91 rhs.second.getTestCaseInfo();
92 }
93 return lhs.first < rhs.first;
94 } );
95
96 std::vector<TestCaseHandle> randomized;
97 randomized.reserve(indexed_tests.size());
98
99 for (auto const& indexed : indexed_tests) {
100 randomized.push_back(indexed.second);
101 }
102
103 return randomized;
104 }
105 }
106
107 CATCH_INTERNAL_ERROR("Unknown test order value!");
108 }

114 std::vector<TestCaseHandle> filterTests( std::vector<TestCaseHandle> const& testCases, TestSpec const& testSpec, IConfig const& config ) {
115 std::vector<TestCaseHandle> filtered;
116 filtered.reserve( testCases.size() );
117 for (auto const& testCase : testCases) {
118 if ((!testSpec.hasFilters() && !testCase.getTestCaseInfo().isHidden()) ||
119 (testSpec.hasFilters() && matchTest(testCase, testSpec, config))) {
120 filtered.push_back(testCase);
121 }
122 }
123 return createShard(filtered, config.shardCount(), config.shardIndex());
124 }

131 // We pre-reserve some reasonable number of tests to avoid the
132 // initial geometric growth churning during test registration.

154 std::vector<TestCaseHandle> const& TestRegistry::getAllTestsSorted( IConfig const& config ) const {
155 if( m_sortedFunctions.empty() )
156 enforceNoDuplicateTestCases( m_handles );
157
158 if( m_currentSortOrder != config.runOrder() || m_sortedFunctions.empty() ) {
159 m_sortedFunctions = sortTests( config, m_handles );
160 m_currentSortOrder = config.runOrder();
161 }
162 return m_sortedFunctions;
