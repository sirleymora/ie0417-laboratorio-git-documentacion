38 Capturer::Capturer( StringRef macroName,
39 SourceLineInfo const& lineInfo,
40 ResultWas::OfType resultType,
41 StringRef names,
42 bool isScoped):
43 m_isScoped(isScoped) {
44 auto trimmed = [&] (size_t start, size_t end) {
45 while (names[start] == ',' || isspace(static_cast<unsigned char>(names[start]))) {
46 ++start;
47 }
48 while (names[end] == ',' || isspace(static_cast<unsigned char>(names[end]))) {
49 --end;
50 }
51 return names.substr(start, end - start + 1);
52 };
53 auto skipq = [&] (size_t start, char quote) {
54 for (auto i = start + 1; i < names.size() ; ++i) {
55 if (names[i] == quote)
56 return i;
57 if (names[i] == '\\')
58 ++i;
59 }
60 CATCH_INTERNAL_ERROR("CAPTURE parsing encountered unmatched quote");
61 };
62
63 size_t start = 0;
64 std::stack<char> openings;
65 for (size_t pos = 0; pos < names.size(); ++pos) {
66 char c = names[pos];
67 switch (c) {
68 case '[':
69 case '{':
70 case '(':
71 // It is basically impossible to disambiguate between
72 // comparison and start of template args in this context
73// case '<':
74 openings.push(c);
75 break;
76 case ']':
77 case '}':
78 case ')':
79// case '>':
80 openings.pop();
81 break;
82 case '"':
83 case '\'':
84 pos = skipq(pos, c);
85 break;
86 case ',':
87 if (start != pos && openings.empty()) {
88 m_messages.emplace_back(macroName, lineInfo, resultType);
89 m_messages.back().message += trimmed(start, pos);
90 m_messages.back().message += " := "_sr;
91 start = pos;
92 }
93 break;
94 default:; // noop
95 }
96 }
97 assert(openings.empty() && "Mismatched openings");
98 m_messages.emplace_back(macroName, lineInfo, resultType);
99 m_messages.back().message += trimmed(start, names.size() - 1);
100 m_messages.back().message += " := "_sr;
101 }
102 Capturer::~Capturer() {
103 assert( m_captured == m_messages.size() );
104 if ( m_isScoped ) {
105 for ( auto const& message : m_messages ) {
106 Detail::popScopedMessage( message.sequence );
107 }

111 void Capturer::captureValue( size_t index, std::string const& value ) {
112 assert( index < m_messages.size() );
113 m_messages[index].message += value;
114 if ( m_isScoped ) {
115 Detail::pushScopedMessage( CATCH_MOVE( m_messages[index] ) );
116 } else {
117 Detail::addUnscopedMessage( CATCH_MOVE( m_messages[index] ) );
118 }
