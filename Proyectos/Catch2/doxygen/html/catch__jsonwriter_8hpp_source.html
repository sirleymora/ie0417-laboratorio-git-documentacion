55 // Without this SFINAE, this overload is a better match
56 // for `std::string`, `char const*`, `char const[N]` args.
57 // While it would still work, it would cause code bloat
58 // and multiple iteration over the strings
59 template <typename T,
60 typename = typename std::enable_if_t<
