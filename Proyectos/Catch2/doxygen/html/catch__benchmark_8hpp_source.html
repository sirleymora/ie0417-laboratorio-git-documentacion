41 template <typename Clock>
42 ExecutionPlan prepare( const IConfig& cfg,
43 Environment env,
44 BenchmarkFunction&& fun ) {
45 auto min_time =
46 env.clock_resolution.mean * Detail::minimum_ticks;
47 auto run_time =
48 std::max( min_time,
49 std::chrono::duration_cast<decltype( min_time )>(
50 cfg.benchmarkWarmupTime() ) );
51 auto&& test = Detail::run_for_at_least<Clock>(
52 std::chrono::duration_cast<IDuration>( run_time ), 1, fun );
53 int new_iters = static_cast<int>(
54 std::ceil( min_time * test.iterations / test.elapsed ) );
55 return { new_iters,
56 test.elapsed / test.iterations * new_iters *
57 cfg.benchmarkSamples(),
58 CATCH_MOVE( fun ),
59 std::chrono::duration_cast<FDuration>(
60 cfg.benchmarkWarmupTime() ),
61 Detail::warmup_iterations };
62 }

130 : fun(CATCH_MOVE(func)), name(CATCH_MOVE(benchmarkName)) {}

132 template <typename Clock = default_clock>
133 void run() {
134 static_assert( Clock::is_steady,
135 "Benchmarking clock should be steady" );
136 auto const* cfg = getCurrentContext().getConfig();
137
138 auto env = Detail::measureEnvironmentDispatch<Clock>();
139
140 getResultCapture().benchmarkPreparing(name);
141 CATCH_TRY{
142 auto plan = user_code([&] {
143 return Detail::prepareDispatch<Clock>( *cfg, env, CATCH_MOVE(fun) );
144 });
145
146 BenchmarkInfo info {
147 CATCH_MOVE(name),
148 plan.estimated_duration.count(),
149 plan.iterations_per_sample,
150 cfg->benchmarkSamples(),
151 cfg->benchmarkResamples(),
152 env.clock_resolution.mean.count(),
153 env.clock_cost.mean.count()
154 };
155
156 getResultCapture().benchmarkStarting(info);
157
158 auto samples = user_code([&] {
159 return Detail::runPlanDispatch<Clock>( plan, *cfg, env );
160 });
161
162 auto analysis = Detail::analyse(*cfg, samples.data(), samples.data() + samples.size());
163 BenchmarkStats<> stats{ CATCH_MOVE(info), CATCH_MOVE(analysis.samples), analysis.mean, analysis.standard_deviation, analysis.outliers, analysis.outlier_variance };
164 getResultCapture().benchmarkEnded(stats);
165 } CATCH_CATCH_ALL {
166 getResultCapture().benchmarkFailed(translateActiveException());
167 // We let the exception go further up so that the
168 // test case is marked as failed.
169 std::rethrow_exception(std::current_exception());
170 }
171 }

173 // sets lambda to be used in fun *and* executes benchmark!
174 template <typename Fun, std::enable_if_t<!Detail::is_related_v<Fun, Benchmark>, int> = 0>
175 Benchmark & operator=(Fun func) {
176 auto const* cfg = getCurrentContext().getConfig();
177 if (!cfg->skipBenchmarks()) {
178 fun = Detail::BenchmarkFunction(func);
179 run();
180 }
181 return *this;
182 }
183
184 explicit operator bool() {
185 return true;
186 }
187
188 private:
189 Detail::BenchmarkFunction fun;
190 std::string name;
191 };
192 }
