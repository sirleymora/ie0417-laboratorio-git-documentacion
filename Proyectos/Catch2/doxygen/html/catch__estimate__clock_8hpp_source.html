27 template <typename Clock>
28 std::vector<double> resolution(int k) {
29 const size_t points = static_cast<size_t>( k + 1 );
30 // To avoid overhead from the branch inside vector::push_back,
31 // we allocate them all and then overwrite.
32 std::vector<TimePoint<Clock>> times(points);
33 for ( auto& time : times ) {
34 time = Clock::now();
35 }
36
37 std::vector<double> deltas;
38 deltas.reserve(static_cast<size_t>(k));
39 for ( size_t idx = 1; idx < points; ++idx ) {
40 deltas.push_back( static_cast<double>(
41 ( times[idx] - times[idx - 1] ).count() ) );
42 }
43
44 return deltas;
45 }

62 template <typename Clock>
63 EnvironmentEstimate estimate_clock_resolution(int iterations) {
64 auto r = run_for_at_least<Clock>(clock_resolution_estimation_time, iterations, &resolution<Clock>)
65 .result;
66 return {
67 FDuration(mean(r.data(), r.data() + r.size())),
68 classify_outliers(r.data(), r.data() + r.size()),
69 };

71 template <typename Clock>
72 EnvironmentEstimate estimate_clock_cost(FDuration resolution) {
73 auto time_limit = (std::min)(
74 resolution * clock_cost_estimation_tick_limit,
75 FDuration(clock_cost_estimation_time_limit));
76 auto time_clock = [](int k) {
77 return Detail::measure<Clock>([k] {
78 for (int i = 0; i < k; ++i) {
79 volatile auto ignored = Clock::now();
80 (void)ignored;
81 }
82 }).elapsed;
83 };
84 time_clock(1);
85 int iters = clock_cost_estimation_iterations;
86 auto&& r = run_for_at_least<Clock>(clock_cost_estimation_time, iters, time_clock);
87 std::vector<double> times;
88 int nsamples = static_cast<int>(std::ceil(time_limit / r.elapsed));
89 times.reserve(static_cast<size_t>(nsamples));
90 for ( int s = 0; s < nsamples; ++s ) {
91 times.push_back( static_cast<double>(
92 ( time_clock( r.iterations ) / r.iterations )
93 .count() ) );
94 }
95 return {
96 FDuration(mean(times.data(), times.data() + times.size())),
97 classify_outliers(times.data(), times.data() + times.size()),
98 };
99 }

101 template <typename Clock>
102 Environment measure_environment() {
103#if defined(__clang__)
104# pragma clang diagnostic push
105# pragma clang diagnostic ignored "-Wexit-time-destructors"
106#endif
107 static Catch::Detail::unique_ptr<Environment> env;
108#if defined(__clang__)
109# pragma clang diagnostic pop
110#endif
111 if (env) {
112 return *env;
113 }
114
115 auto iters = Detail::warmup<Clock>();
116 auto resolution = Detail::estimate_clock_resolution<Clock>(iters);
117 auto cost = Detail::estimate_clock_cost<Clock>(resolution.mean);
118
119 env = Catch::Detail::make_unique<Environment>( Environment{resolution, cost} );
120 return *env;
121 }
