23 template <typename TargetRangeLike, typename Equality>
24 class RangeEqualsMatcher final : public MatcherGenericBase {
25 TargetRangeLike m_desired;
26 Equality m_predicate;
27
28 public:
29 template <typename TargetRangeLike2, typename Equality2>
30 constexpr
31 RangeEqualsMatcher( TargetRangeLike2&& range,
32 Equality2&& predicate ):
33 m_desired( CATCH_FORWARD( range ) ),
34 m_predicate( CATCH_FORWARD( predicate ) ) {}
35
36 template <typename RangeLike>
37 constexpr
38 bool match( RangeLike&& rng ) const {
39 auto rng_start = begin( rng );
40 const auto rng_end = end( rng );
41 auto target_start = begin( m_desired );
42 const auto target_end = end( m_desired );
43
44 while (rng_start != rng_end && target_start != target_end) {
45 if (!m_predicate(*rng_start, *target_start)) {
46 return false;
47 }
48 ++rng_start;
49 ++target_start;
50 }
51 return rng_start == rng_end && target_start == target_end;
52 }
53
54 std::string describe() const override {
55 return "elements are " + Catch::Detail::stringify( m_desired );
56 }
57 };
