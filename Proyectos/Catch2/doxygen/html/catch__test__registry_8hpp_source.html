31template<typename C>
32class TestInvokerAsMethod final : public ITestInvoker {
33 void (C::*m_testAsMethod)();
34public:
35 constexpr TestInvokerAsMethod( void ( C::*testAsMethod )() ) noexcept:
36 m_testAsMethod( testAsMethod ) {}
37
38 void invoke() const override {
39 C obj;

51template <typename C>
52class TestInvokerFixture final : public ITestInvoker {
53 void ( C::*m_testAsMethod )() const;
54 Detail::unique_ptr<C> m_fixture = nullptr;
55
56public:
57 constexpr TestInvokerFixture( void ( C::*testAsMethod )() const ) noexcept:
58 m_testAsMethod( testAsMethod ) {}
59
60 void prepareTestCase() override {
61 m_fixture = Detail::make_unique<C>();
62 }
63
64 void tearDownTestCase() override {
65 m_fixture.reset();
66 }
67
68 void invoke() const override {
69 auto* f = m_fixture.get();
70 ( f->*m_testAsMethod )();
71 }
72};
