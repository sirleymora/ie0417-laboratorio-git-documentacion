24 static StringRef extractClassName( StringRef classOrMethodName ) {
25 if ( !startsWith( classOrMethodName, '&' ) ) {
26 return classOrMethodName;
27 }
28
29 // Remove the leading '&' to avoid having to special case it later
30 const auto methodName =
31 classOrMethodName.substr( 1, classOrMethodName.size() );
32
33 auto reverseStart = std::make_reverse_iterator( methodName.end() );
34 auto reverseEnd = std::make_reverse_iterator( methodName.begin() );
35
36 // We make a simplifying assumption that ":" is only present
37 // in the input as part of "::" from C++ typenames (this is
38 // relatively safe assumption because the input is generated
39 // as stringification of type through preprocessor).
40 auto lastColons = std::find( reverseStart, reverseEnd, ':' ) + 1;
41 auto secondLastColons =
42 std::find( lastColons + 1, reverseEnd, ':' );
43
44 auto const startIdx = reverseEnd - secondLastColons;
45 auto const classNameSize = secondLastColons - lastColons - 1;
46
47 return methodName.substr(
48 static_cast<std::size_t>( startIdx ),
49 static_cast<std::size_t>( classNameSize ) );
50 }
