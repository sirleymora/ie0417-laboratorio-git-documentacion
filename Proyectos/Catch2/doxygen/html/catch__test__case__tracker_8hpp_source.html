26 struct NameAndLocation {
27 std::string name;
28 SourceLineInfo location;
29
30 NameAndLocation( std::string&& _name, SourceLineInfo const& _location );
31 friend bool operator==(NameAndLocation const& lhs, NameAndLocation const& rhs) {
32 // This is a very cheap check that should have a very high hit rate.
33 // If we get to SourceLineInfo::operator==, we will redo it, but the
34 // cost of repeating is trivial at that point (we will be paying
35 // multiple strcmp/memcmps at that point).
36 if ( lhs.location.line != rhs.location.line ) { return false; }
37 return lhs.name == rhs.name && lhs.location == rhs.location;
38 }
39 friend bool operator!=(NameAndLocation const& lhs,
40 NameAndLocation const& rhs) {
41 return !( lhs == rhs );
42 }
43 };

60 friend bool operator==( NameAndLocation const& lhs,

62 // This is a very cheap check that should have a very high hit rate.
63 // If we get to SourceLineInfo::operator==, we will redo it, but the
64 // cost of repeating is trivial at that point (we will be paying
65 // multiple strcmp/memcmps at that point).
66 if ( lhs.location.line != rhs.location.line ) { return false; }
67 return StringRef( lhs.name ) == rhs.name &&
68 lhs.location == rhs.location;
69 }
70 friend bool operator==( NameAndLocationRef const& lhs,
71 NameAndLocation const& rhs ) {

104 // Note: There are 2 dummy section trackers (root, test-case) before
