27 // This is a bunch of helpers for the templated test case handling.
28 // They should live elsewhere in the long run, but as an in-between

30 template <typename...> struct TypeList {};
31 template <typename... Ts>
32 constexpr auto get_wrapper( priority_tag<1> ) noexcept -> TypeList<Ts...> { return {}; }
33 template <template <typename...> class...> struct TemplateTypeList {};
34 // Clang 20 and 21 cannot handle an explicitly specified all-pack
35 // template-template parameter here ("conflicting deduction" regression,
36 // llvm/llvm-project#130778; fixed for Clang 22).
37 // Remove get_template_wrapper once Clang 21 is no longer supported.
38 template <template <typename...> class C, template <typename...> class... Cs>
39 constexpr auto get_template_wrapper( priority_tag<1> ) noexcept -> TemplateTypeList<C, Cs...> { return {}; }

41 template <typename...>
42 struct append;
43 template <typename T>
44 struct append<T> { using type = T; };
45 template <template <typename...> class L1, typename... E1, template <typename...> class L2, typename... E2, typename... Rest>
46 struct append<L1<E1...>, L2<E2...>, Rest...> { using type = typename append<L1<E1..., E2...>, Rest...>::type; };
47 template <template <typename...> class L1, typename... E1, typename... Rest>
48 struct append<L1<E1...>, TypeList<mpl_::na>, Rest...> { using type = L1<E1...>; };

50 template <template <typename...> class, typename>
51 struct convert;
52 template <template <typename...> class Final, template <typename...> class List, typename... Ts>
53 struct convert<Final, List<Ts...>> { using type = typename append<Final<>, TypeList<Ts>...>::type; };

55 // These are helpers for the PRODUCT templated test cases.
56 // Note that the _SIG macros (for NTTPs) also use specializations
57 // of these, but they have to use their own instances due to needing
58 // per-sig specializations and we have to keep these in their own
59 // unnamed namespace.
60 template <typename...>
61 struct rewrap;
62 template <template <typename...> class Container, template <typename...> class List, typename... elems>
63 struct rewrap<TemplateTypeList<Container>, List<elems...>> { using type = TypeList<Container<elems...>>; };
64 template <template <typename...> class Container, template <typename...> class List, class... Elems, typename... Elements>
65 struct rewrap<TemplateTypeList<Container>, List<Elems...>, Elements...> { using type = typename append<TypeList<Container<Elems...>>, typename rewrap<TemplateTypeList<Container>, Elements...>::type>::type; };
66
67 template <template <typename...> class, typename...>
68 struct create;
69 template <template <typename...> class Final, template <typename...> class... Containers, typename... Types>
70 struct create<Final, TemplateTypeList<Containers...>, TypeList<Types...>> { using type = typename append<Final<>, typename rewrap<TemplateTypeList<Containers>, Types...>::type...>::type; };
71 }
