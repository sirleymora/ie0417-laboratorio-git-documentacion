46 void AssertionHandler::complete() {
47 m_completed = true;
48 if( m_reaction.shouldDebugBreak ) {
49
50 // If you find your debugger stopping you here then go one level up on the
51 // call-stack for the code that caused it (typically a failed assertion)
52
53 // (To go back to the test and change execution, jump over the throw, next)
54 CATCH_BREAK_INTO_DEBUGGER();
55 }
56 if (m_reaction.shouldThrow) {
57 throw_test_failure_exception();
58 }
59 if ( m_reaction.shouldSkip ) {
60 throw_test_skip_exception();
61 }
62 }
