45class ConsoleAssertionPrinter {
46public:
47 ConsoleAssertionPrinter& operator= (ConsoleAssertionPrinter const&) = delete;
48 ConsoleAssertionPrinter(ConsoleAssertionPrinter const&) = delete;
49 ConsoleAssertionPrinter(std::ostream& _stream, AssertionStats const& _stats, ColourImpl* colourImpl_, bool _printInfoMessages)
50 : stream(_stream),
51 stats(_stats),
52 result(_stats.assertionResult),
53 colour(Colour::None),
54 messages(_stats.infoMessages),
55 colourImpl(colourImpl_),
56 printInfoMessages(_printInfoMessages) {
57 switch (result.getResultType()) {
58 case ResultWas::Ok:
59 colour = Colour::Success;
60 passOrFail = "PASSED"_sr;
61 //if( result.hasMessage() )
62 if (messages.size() == 1)
63 messageLabel = "with message"_sr;
64 if (messages.size() > 1)
65 messageLabel = "with messages"_sr;
66 break;
67 case ResultWas::ExpressionFailed:
68 if (result.isOk()) {
69 colour = Colour::Success;
70 passOrFail = "FAILED - but was ok"_sr;
71 } else {
72 colour = Colour::Error;
73 passOrFail = "FAILED"_sr;
74 }
75 if (messages.size() == 1)
76 messageLabel = "with message"_sr;
77 if (messages.size() > 1)
78 messageLabel = "with messages"_sr;
79 break;
80 case ResultWas::ThrewException:
81 colour = Colour::Error;
82 passOrFail = "FAILED"_sr;
83 // todo switch
84 switch (messages.size()) { case 0:
85 messageLabel = "due to unexpected exception with "_sr;
86 break;
87 case 1:
88 messageLabel = "due to unexpected exception with message"_sr;
89 break;
90 default:
91 messageLabel = "due to unexpected exception with messages"_sr;
92 break;
93 }
94 break;
95 case ResultWas::FatalErrorCondition:
96 colour = Colour::Error;
97 passOrFail = "FAILED"_sr;
98 messageLabel = "due to a fatal error condition"_sr;
99 break;
100 case ResultWas::DidntThrowException:
101 colour = Colour::Error;
102 passOrFail = "FAILED"_sr;
103 messageLabel = "because no exception was thrown where one was expected"_sr;
104 break;
105 case ResultWas::Info:
106 messageLabel = "info"_sr;
107 break;
108 case ResultWas::Warning:
109 messageLabel = "warning"_sr;
110 break;
111 case ResultWas::ExplicitFailure:
112 passOrFail = "FAILED"_sr;
113 colour = Colour::Error;
114 if (messages.size() == 1)
115 messageLabel = "explicitly with message"_sr;
116 if (messages.size() > 1)
117 messageLabel = "explicitly with messages"_sr;
118 break;
119 case ResultWas::ExplicitSkip:
120 colour = Colour::Skip;
121 passOrFail = "SKIPPED"_sr;
122 if (messages.size() == 1)
123 messageLabel = "explicitly with message"_sr;
124 if (messages.size() > 1)
125 messageLabel = "explicitly with messages"_sr;
126 break;
127 // These cases are here to prevent compiler warnings
128 case ResultWas::Unknown:
129 case ResultWas::FailureBit:
130 case ResultWas::Exception:
131 passOrFail = "** internal error **"_sr;
132 colour = Colour::Error;
133 break;
134 }
135 }

137 void print() const {
138 printSourceInfo();
139 if (stats.totals.assertions.total() > 0) {
140 printResultType();
141 printOriginalExpression();
142 printReconstructedExpression();
143 } else {
144 stream << '\n';
145 }
146 printMessage();
147 }
148
149private:
150 void printResultType() const {
151 if (!passOrFail.empty()) {
152 stream << colourImpl->guardColour(colour) << passOrFail << ":\n";
153 }
154 }
155 void printOriginalExpression() const {
156 if (result.hasExpression()) {
157 stream << colourImpl->guardColour( Colour::OriginalExpression )
158 << " " << result.getExpressionInMacro() << '\n';
159 }
160 }
161 void printReconstructedExpression() const {
162 if (result.hasExpandedExpression()) {
163 stream << "with expansion:\n";
164 stream << colourImpl->guardColour( Colour::ReconstructedExpression )
165 << TextFlow::Column( result.getExpandedExpression() )
166 .indent( 2 )
167 << '\n';
168 }
169 }
170 void printMessage() const {
171 if (!messageLabel.empty())
172 stream << messageLabel << ':' << '\n';
173 for (auto const& msg : messages) {
174 // If this assertion is a warning ignore any INFO messages
175 if (printInfoMessages || msg.type != ResultWas::Info)
176 stream << TextFlow::Column(msg.message).indent(2) << '\n';
177 }
178 }
179 void printSourceInfo() const {
180 stream << colourImpl->guardColour( Colour::FileName )
181 << result.getSourceInfo() << ": ";
182 }

184 std::ostream& stream;
185 AssertionStats const& stats;
186 AssertionResult const& result;
187 Colour::Code colour;
188 StringRef passOrFail;
189 StringRef messageLabel;
190 std::vector<MessageInfo> const& messages;
191 ColourImpl* colourImpl;
192 bool printInfoMessages;
193};

200std::size_t&
201findMax( std::size_t& i, std::size_t& j, std::size_t& k, std::size_t& l ) {
202 if (i > j && i > k && i > l)
203 return i;
204 else if (j > k && j > l)
205 return j;
206 else if (k > l)
207 return k;
208 else
209 return l;

216class Duration {
217 enum class Unit : uint8_t {
218 Auto,
219 Nanoseconds,
220 Microseconds,
221 Milliseconds,
222 Seconds,
223 Minutes
224 };
225 static const uint64_t s_nanosecondsInAMicrosecond = 1000;
226 static const uint64_t s_nanosecondsInAMillisecond = 1000 * s_nanosecondsInAMicrosecond;
227 static const uint64_t s_nanosecondsInASecond = 1000 * s_nanosecondsInAMillisecond;
228 static const uint64_t s_nanosecondsInAMinute = 60 * s_nanosecondsInASecond;
229
230 double m_inNanoseconds;
231 Unit m_units;
232
233public:
234 explicit Duration(double inNanoseconds, Unit units = Unit::Auto)
235 : m_inNanoseconds(inNanoseconds),
236 m_units(units) {
237 if (m_units == Unit::Auto) {
238 if (m_inNanoseconds < s_nanosecondsInAMicrosecond)
239 m_units = Unit::Nanoseconds;
240 else if (m_inNanoseconds < s_nanosecondsInAMillisecond)
241 m_units = Unit::Microseconds;
242 else if (m_inNanoseconds < s_nanosecondsInASecond)
243 m_units = Unit::Milliseconds;
244 else if (m_inNanoseconds < s_nanosecondsInAMinute)
245 m_units = Unit::Seconds;
246 else
247 m_units = Unit::Minutes;
248 }
249
250 }
251
252 auto value() const -> double {
253 switch (m_units) {
254 case Unit::Microseconds:
255 return m_inNanoseconds / static_cast<double>(s_nanosecondsInAMicrosecond);
256 case Unit::Milliseconds:
257 return m_inNanoseconds / static_cast<double>(s_nanosecondsInAMillisecond);
258 case Unit::Seconds:
259 return m_inNanoseconds / static_cast<double>(s_nanosecondsInASecond);
260 case Unit::Minutes:
261 return m_inNanoseconds / static_cast<double>(s_nanosecondsInAMinute);
262 default:
263 return m_inNanoseconds;
264 }
265 }
266 StringRef unitsAsString() const {
267 switch (m_units) {
268 case Unit::Nanoseconds:
269 return "ns"_sr;
270 case Unit::Microseconds:
271 return "us"_sr;
272 case Unit::Milliseconds:
273 return "ms"_sr;
274 case Unit::Seconds:
275 return "s"_sr;
276 case Unit::Minutes:
277 return "m"_sr;
278 default:
279 return "** internal error **"_sr;
280 }
281
282 }
283 friend auto operator << (std::ostream& os, Duration const& duration) -> std::ostream& {
284 return os << duration.value() << ' ' << duration.unitsAsString();
285 }

300class TablePrinter {
301 std::ostream& m_os;
302 std::vector<ColumnInfo> m_columnInfos;
303 ReusableStringStream m_oss;
304 int m_currentColumn = -1;
305 bool m_isOpen = false;
306
307public:
308 TablePrinter( std::ostream& os, std::vector<ColumnInfo> columnInfos )
309 : m_os( os ),
310 m_columnInfos( CATCH_MOVE( columnInfos ) ) {}
311
312 auto columnInfos() const -> std::vector<ColumnInfo> const& {
313 return m_columnInfos;
314 }
315
316 void open() {
317 if (!m_isOpen) {
318 m_isOpen = true;
319 *this << RowBreak();
320
321 TextFlow::Columns headerCols;
322 for (auto const& info : m_columnInfos) {
323 assert(info.width > 2);
324 headerCols += TextFlow::Column(info.name).width(info.width - 2);
325 headerCols += TextFlow::Spacer( 2 );
326 }
327 m_os << headerCols << '\n';
328
329 m_os << lineOfChars('-') << '\n';
330 }
331 }
332 void close() {
333 if (m_isOpen) {
334 *this << RowBreak();
335 m_os << '\n' << std::flush;
336 m_isOpen = false;
337 }
338 }
339
340 template<typename T>
341 friend TablePrinter& operator<< (TablePrinter& tp, T const& value) {
342 tp.m_oss << value;
343 return tp;
344 }
345
346 friend TablePrinter& operator<< (TablePrinter& tp, ColumnBreak) {
347 auto colStr = tp.m_oss.str();
348 const auto strSize = colStr.size();
349 tp.m_oss.str("");
350 tp.open();
351 if (tp.m_currentColumn == static_cast<int>(tp.m_columnInfos.size() - 1)) {
352 tp.m_currentColumn = -1;
353 tp.m_os << '\n';
354 }
355 tp.m_currentColumn++;
356
357 auto colInfo = tp.m_columnInfos[tp.m_currentColumn];
358 auto padding = (strSize + 1 < colInfo.width)
359 ? std::string(colInfo.width - (strSize + 1), ' ')
360 : std::string();
361 if (colInfo.justification == Justification::Left)
362 tp.m_os << colStr << padding << ' ';
363 else
364 tp.m_os << padding << colStr << ' ';
365 return tp;
366 }
367
368 friend TablePrinter& operator<< (TablePrinter& tp, RowBreak) {
369 if (tp.m_currentColumn > 0) {
370 tp.m_os << '\n';
371 tp.m_currentColumn = -1;
372 }
373 return tp;
374 }
375
376 friend TablePrinter& operator<<(TablePrinter& tp, OutputFlush) {
377 tp.m_os << std::flush;
378 return tp;
379 }

598void ConsoleReporter::printHeaderString(std::string const& _string, std::size_t indent) {
599 // We want to get a bit fancy with line breaking here, so that subsequent
600 // lines start after ":" if one is present, e.g.
601 // ```
602 // blablabla: Fancy
603 // linebreaking
604 // ```
605 // but we also want to avoid problems with overly long indentation causing
606 // the text to take up too many lines, e.g.
607 // ```
608 // blablabla: F
609 // a
610 // n
611 // c
612 // y
613 // .
614 // .
615 // .
616 // ```
617 // So we limit the prefix indentation check to first quarter of the possible
618 // width
619 std::size_t idx = _string.find( ": " );
620 if ( idx != std::string::npos && idx < CATCH_CONFIG_CONSOLE_WIDTH / 4 ) {
621 idx += 2;
622 } else {
623 idx = 0;
624 }
625 m_stream << TextFlow::Column( _string )
626 .indent( indent + idx )
627 .initialIndent( indent )
628 << '\n';
629}

631void ConsoleReporter::printTotalsDivider(Totals const& totals) {
632 if (totals.testCases.total() > 0) {
633 std::size_t failedRatio = makeRatio(totals.testCases.failed, totals.testCases.total());
634 std::size_t failedButOkRatio = makeRatio(totals.testCases.failedButOk, totals.testCases.total());
635 std::size_t passedRatio = makeRatio(totals.testCases.passed, totals.testCases.total());
636 std::size_t skippedRatio = makeRatio(totals.testCases.skipped, totals.testCases.total());
637 while (failedRatio + failedButOkRatio + passedRatio + skippedRatio < CATCH_CONFIG_CONSOLE_WIDTH - 1)
638 findMax(failedRatio, failedButOkRatio, passedRatio, skippedRatio)++;
639 while (failedRatio + failedButOkRatio + passedRatio > CATCH_CONFIG_CONSOLE_WIDTH - 1)
640 findMax(failedRatio, failedButOkRatio, passedRatio, skippedRatio)--;
641
642 m_stream << m_colour->guardColour( Colour::Error )
643 << std::string( failedRatio, '=' )
644 << m_colour->guardColour( Colour::ResultExpectedFailure )
645 << std::string( failedButOkRatio, '=' );
646 if ( totals.testCases.allPassed() ) {
647 m_stream << m_colour->guardColour( Colour::ResultSuccess )
648 << std::string( passedRatio, '=' );
649 } else {
650 m_stream << m_colour->guardColour( Colour::Success )
651 << std::string( passedRatio, '=' );
652 }
653 m_stream << m_colour->guardColour( Colour::Skip )
654 << std::string( skippedRatio, '=' );
655 } else {
656 m_stream << m_colour->guardColour( Colour::Warning )
657 << std::string( CATCH_CONFIG_CONSOLE_WIDTH - 1, '=' );
658 }
659 m_stream << '\n';
660}
