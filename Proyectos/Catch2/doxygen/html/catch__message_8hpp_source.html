105#define INTERNAL_CATCH_MSG( macroName, messageType, resultDisposition, ... ) \
106 do { \
107 Catch::AssertionHandler catchAssertionHandler( macroName##_catch_sr, CATCH_INTERNAL_LINEINFO, Catch::StringRef(), resultDisposition ); \
108 catchAssertionHandler.handleMessage( messageType, ( Catch::MessageStream() << __VA_ARGS__ + ::Catch::StreamEndStop() ).m_stream.str() ); \
109 catchAssertionHandler.complete(); \
