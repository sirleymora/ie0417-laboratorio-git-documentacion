23 ExecutionPlan prepare_default( const IConfig& cfg,
24 Environment env,
25 BenchmarkFunction&& fun ) {
26 // This mirrors Benchmark::prepare<default_clock>(), but with the
27 // clock fixed so it is instantiated once here in the library.
28 auto min_time = env.clock_resolution.mean * Detail::minimum_ticks;
29 auto run_time = std::max(
30 min_time,
31 std::chrono::duration_cast<decltype( min_time )>(
32 cfg.benchmarkWarmupTime() ) );
33 auto&& test = Detail::run_for_at_least<default_clock>(
34 std::chrono::duration_cast<IDuration>( run_time ), 1, fun );
35 int new_iters = static_cast<int>(
36 std::ceil( min_time * test.iterations / test.elapsed ) );
37 return { new_iters,
38 test.elapsed / test.iterations * new_iters *
39 cfg.benchmarkSamples(),
40 CATCH_MOVE( fun ),
41 std::chrono::duration_cast<FDuration>(
42 cfg.benchmarkWarmupTime() ),
43 Detail::warmup_iterations };
44 }

46 std::vector<FDuration> run_plan_default( ExecutionPlan const& plan,
47 const IConfig& cfg,
48 Environment env ) {
49 return plan.run<default_clock>( cfg, env );
50 }
