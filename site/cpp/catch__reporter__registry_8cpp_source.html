57 IEventListenerPtr
58 ReporterRegistry::create( std::string const& name,
59 ReporterConfig&& config ) const {
60 auto it = m_impl->factories.find( name );
61 if ( it == m_impl->factories.end() ) return nullptr;
62 return it->second->create( CATCH_MOVE( config ) );

65 void ReporterRegistry::registerReporter( std::string const& name,
66 IReporterFactoryPtr factory ) {
67 CATCH_ENFORCE( name.find( "::" ) == name.npos,
68 "'::' is not allowed in reporter name: '" + name +
69 '\'' );
70 auto ret = m_impl->factories.emplace( name, CATCH_MOVE( factory ) );
71 CATCH_ENFORCE( ret.second,
72 "reporter using '" + name +
73 "' as name was already registered" );
74 }
75 void ReporterRegistry::registerListener(
76 Detail::unique_ptr<EventListenerFactory> factory ) {
77 m_impl->listeners.push_back( CATCH_MOVE( factory ) );
78 }
