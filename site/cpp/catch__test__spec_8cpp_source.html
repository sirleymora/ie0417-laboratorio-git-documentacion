62 bool TestSpec::Filter::matches( TestCaseInfo const& testCase ) const {
63 bool should_use = !testCase.isHidden();
64 for (auto const& pattern : m_required) {
65 should_use = true;
66 if (!pattern->matches(testCase)) {
67 return false;
68 }
69 }
70 for (auto const& pattern : m_forbidden) {
71 if (pattern->matches(testCase)) {
72 return false;
73 }
74 }

111 TestSpec::Matches TestSpec::matchesByFilter( std::vector<TestCaseHandle> const& testCases, IConfig const& config ) const {
112 Matches matches;
113 matches.reserve( m_filters.size() );
114 for ( auto const& filter : m_filters ) {
115 std::vector<TestCaseHandle const*> currentMatches;
116 for ( auto const& test : testCases )
117 if ( isThrowSafe( test, config ) &&
118 filter.matches( test.getTestCaseInfo() ) )
119 currentMatches.emplace_back( &test );
120 matches.push_back(
121 FilterMatch{ extractFilterName( filter ), currentMatches } );
122 }
123 return matches;
124 }
