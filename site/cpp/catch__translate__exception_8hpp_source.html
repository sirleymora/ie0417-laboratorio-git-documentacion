23 class ExceptionTranslatorRegistrar {
24 template<typename T>
25 class ExceptionTranslator final : public IExceptionTranslator {
26 public:
27
28 constexpr ExceptionTranslator( std::string(*translateFunction)( T const& ) )
29 : m_translateFunction( translateFunction )
30 {}
31
32 std::string translate( ExceptionTranslators::const_iterator it, ExceptionTranslators::const_iterator itEnd ) const override {
33#if !defined(CATCH_CONFIG_DISABLE_EXCEPTIONS)
34 try {
35 if( it == itEnd )
36 std::rethrow_exception(std::current_exception());
37 else
38 return (*it)->translate( it+1, itEnd );
39 }
40 catch( T const& ex ) {
41 return m_translateFunction( ex );
42 }
43#else
44 (void)it;
45 (void)itEnd;
46 return "You should never get here!";
47#endif
48 }
49
50 protected:
51 std::string(*m_translateFunction)( T const& );
52 };
53
54 public:
55 template<typename T>
56 ExceptionTranslatorRegistrar( std::string(*translateFunction)( T const& ) ) {
57 Detail::registerTranslatorImpl(
58 Detail::make_unique<ExceptionTranslator<T>>(
59 translateFunction ) );
60 }
61 };
