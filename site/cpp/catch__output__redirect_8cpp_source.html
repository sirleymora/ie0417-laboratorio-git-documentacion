103 // Windows's implementation of std::tmpfile is terrible (it tries
104 // to create a file inside system folder, thus requiring elevated
105 // privileges for the binary), so we have to use tmpnam(_s) and
106 // create the file ourselves there.
107 class TempFile {
108 public:
109 TempFile( TempFile const& ) = delete;
110 TempFile& operator=( TempFile const& ) = delete;
111 TempFile( TempFile&& ) = delete;
112 TempFile& operator=( TempFile&& ) = delete;
113
114# if defined( _MSC_VER )
115 TempFile() {
116 if ( tmpnam_s( m_buffer ) ) {
117 CATCH_RUNTIME_ERROR( "Could not get a temp filename" );
118 }
119 if ( fopen_s( &m_file, m_buffer, "wb+" ) ) {
120 char buffer[100];
121 if ( strerror_s( buffer, errno ) ) {
122 CATCH_RUNTIME_ERROR(
123 "Could not translate errno to a string" );
124 }
125 CATCH_RUNTIME_ERROR( "Could not open the temp file: '"
126 << m_buffer
127 << "' because: " << buffer );
128 }
129 }
130# else
131 TempFile() {
132 m_file = std::tmpfile();
133 if ( !m_file ) {
134 CATCH_RUNTIME_ERROR( "Could not create a temp file." );
135 }
136 }
137# endif
138
139 ~TempFile() {
140 // TBD: What to do about errors here?
141 std::fclose( m_file );
142 // We manually create the file on Windows only, on Linux
143 // it will be autodeleted
144# if defined( _MSC_VER )
145 std::remove( m_buffer );
146# endif
147 }
148
149 std::FILE* getFile() { return m_file; }
150 std::string getContents() {
151 ReusableStringStream sstr;
152 constexpr long buffer_size = 100;
153 char buffer[buffer_size + 1] = {};
154 long current_pos = ftell( m_file );
155 CATCH_ENFORCE( current_pos >= 0,
156 "ftell failed, errno: " << errno );
157 std::rewind( m_file );
158 while ( current_pos > 0 ) {
159 auto read_characters =
160 std::fread( buffer,
161 1,
162 std::min( buffer_size, current_pos ),
163 m_file );
164 buffer[read_characters] = '\0';
165 sstr << buffer;
166 current_pos -= static_cast<long>( read_characters );
167 }
168 return sstr.str();
169 }
170
171 void clear() { std::rewind( m_file ); }
172
173 private:
174 std::FILE* m_file = nullptr;
175 char m_buffer[L_tmpnam] = { 0 };
176 };

184 class FileRedirect final : public OutputRedirect {
185 TempFile m_outFile, m_errFile;
186 int m_originalOut = -1;
187 int m_originalErr = -1;
188
189 // Flushes cout/cerr/clog streams and stdout/stderr FDs
190 void flushEverything() {
191 Catch::cout() << std::flush;
192 fflush( stdout );
193 // Since we support overriding these streams, we flush cerr
194 // even though std::cerr is unbuffered
195 Catch::cerr() << std::flush;
196 Catch::clog() << std::flush;
197 fflush( stderr );
198 }
199
200 public:
201 FileRedirect():
202 m_originalOut( dup( fileno( stdout ) ) ),
203 m_originalErr( dup( fileno( stderr ) ) ) {
204 CATCH_ENFORCE( m_originalOut >= 0, "Could not dup stdout" );
205 CATCH_ENFORCE( m_originalErr >= 0, "Could not dup stderr" );
206 }
207
208 std::string getStdout() override { return m_outFile.getContents(); }
209 std::string getStderr() override { return m_errFile.getContents(); }
210 void clearBuffers() override {
211 m_outFile.clear();
212 m_errFile.clear();
213 }
214
215 void activateImpl() override {
216 // We flush before starting redirect, to ensure that we do
217 // not capture the end of message sent before activation.
218 flushEverything();
219
220 int ret;
221 ret = dup2( fileno( m_outFile.getFile() ), fileno( stdout ) );
222 CATCH_ENFORCE( ret >= 0,
223 "dup2 to stdout has failed, errno: " << errno );
224 ret = dup2( fileno( m_errFile.getFile() ), fileno( stderr ) );
225 CATCH_ENFORCE( ret >= 0,
226 "dup2 to stderr has failed, errno: " << errno );
227 }
228 void deactivateImpl() override {
229 // We flush before ending redirect, to ensure that we
230 // capture all messages sent while the redirect was active.
231 flushEverything();
232
233 int ret;
234 ret = dup2( m_originalOut, fileno( stdout ) );
235 CATCH_ENFORCE(
236 ret >= 0,
237 "dup2 of original stdout has failed, errno: " << errno );
238 ret = dup2( m_originalErr, fileno( stderr ) );
239 CATCH_ENFORCE(
240 ret >= 0,
241 "dup2 of original stderr has failed, errno: " << errno );
242 }
243 };

287 RedirectGuard::RedirectGuard( bool activate, OutputRedirect& redirectImpl ):
288 m_redirect( &redirectImpl ),
289 m_activate( activate ),
290 m_previouslyActive( redirectImpl.isActive() ) {
291
292 // Skip cases where there is no actual state change.
293 if ( m_activate == m_previouslyActive ) { return; }
294
295 if ( m_activate ) {
296 m_redirect->activate();
297 } else {
298 m_redirect->deactivate();
299 }
300 }
301
302 RedirectGuard::~RedirectGuard() noexcept( false ) {
303 if ( m_moved ) { return; }
304 // Skip cases where there is no actual state change.
305 if ( m_activate == m_previouslyActive ) { return; }
306
307 if ( m_activate ) {
308 m_redirect->deactivate();
309 } else {
310 m_redirect->activate();
311 }
312 }
