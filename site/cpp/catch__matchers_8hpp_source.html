48 template<typename ArgT>
49 class MatchAllOf final : public MatcherBase<ArgT> {
50 std::vector<MatcherBase<ArgT> const*> m_matchers;
51
52 public:
53 MatchAllOf() = default;
54 MatchAllOf(MatchAllOf const&) = delete;
55 MatchAllOf& operator=(MatchAllOf const&) = delete;
56 MatchAllOf(MatchAllOf&&) = default;
57 MatchAllOf& operator=(MatchAllOf&&) = default;
58
59
60 bool match( ArgT const& arg ) const override {
61 for( auto matcher : m_matchers ) {
62 if (!matcher->match(arg))
63 return false;
64 }
65 return true;
66 }
67 std::string describe() const override {
68 std::string description;
69 description.reserve( 4 + m_matchers.size()*32 );
70 description += "( ";
71 bool first = true;
72 for( auto matcher : m_matchers ) {
73 if( first )
74 first = false;
75 else
76 description += " and ";
77 description += matcher->toString();
78 }
79 description += " )";
80 return description;
81 }
82
83 friend MatchAllOf operator&&( MatchAllOf&& lhs,
84 MatcherBase<ArgT> const& rhs
85 CATCH_ATTR_LIFETIMEBOUND ) {
86 lhs.m_matchers.push_back(&rhs);
87 return CATCH_MOVE(lhs);
88 }
89 friend MatchAllOf
90 operator&&( MatcherBase<ArgT> const& lhs CATCH_ATTR_LIFETIMEBOUND,
91 MatchAllOf&& rhs ) {
92 rhs.m_matchers.insert(rhs.m_matchers.begin(), &lhs);
93 return CATCH_MOVE(rhs);
94 }
95 };

162 template<typename ArgT>
163 class MatchNotOf final : public MatcherBase<ArgT> {
164 MatcherBase<ArgT> const& m_underlyingMatcher;
165
166 public:
167 explicit MatchNotOf( MatcherBase<ArgT> const& underlyingMatcher
168 CATCH_ATTR_LIFETIMEBOUND ):
169 m_underlyingMatcher( underlyingMatcher )
170 {}
171
172 bool match( ArgT const& arg ) const override {
173 return !m_underlyingMatcher.match( arg );
174 }
175
176 std::string describe() const override {
177 return "not " + m_underlyingMatcher.toString();
178 }
179 };
