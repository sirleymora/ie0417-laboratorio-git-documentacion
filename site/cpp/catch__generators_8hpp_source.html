87 template<typename T>
88 class FixedValuesGenerator final : public IGenerator<T> {
89 static_assert(!std::is_same<T, bool>::value,
90 "FixedValuesGenerator does not support bools because of std::vector<bool>"
91 "specialization, use SingleValue Generator instead.");
92 std::vector<T> m_values;
93 size_t m_idx = 0;
94
95 void skipToNthElementImpl( std::size_t n ) override {
96 if ( n >= m_values.size() ) {
97 Detail::throw_generator_exception(
98 "Could not jump to Nth element: not enough elements" );
99 }
100 m_idx = n;
101 }
102
103 public:
104 FixedValuesGenerator( std::initializer_list<T> values ) : m_values( values ) {}
105
106 T const& get() const override {
107 return m_values[m_idx];
108 }
109 bool next() override {
110 ++m_idx;
111 return m_idx < m_values.size();
112 }
113
114 bool isFinite() const override { return true; }
115 };

128 template<typename T>
129 class Generators : public IGenerator<T> {
130 std::vector<GeneratorWrapper<T>> m_generators;
131 size_t m_current = 0;
132
133 void add_generator( GeneratorWrapper<T>&& generator ) {
134 m_generators.emplace_back( CATCH_MOVE( generator ) );
135 }
136 void add_generator( T const& val ) {
137 m_generators.emplace_back( value( val ) );
138 }
139 void add_generator( T&& val ) {
140 m_generators.emplace_back( value( CATCH_MOVE( val ) ) );
141 }
142 template <typename U>
143 std::enable_if_t<!std::is_same<std::decay_t<U>, T>::value>
144 add_generator( U&& val ) {
145 add_generator( T( CATCH_FORWARD( val ) ) );
146 }
147
148 template <typename U> void add_generators( U&& valueOrGenerator ) {
149 add_generator( CATCH_FORWARD( valueOrGenerator ) );
150 }
151
152 template <typename U, typename... Gs>
153 void add_generators( U&& valueOrGenerator, Gs&&... moreGenerators ) {
154 add_generator( CATCH_FORWARD( valueOrGenerator ) );
155 add_generators( CATCH_FORWARD( moreGenerators )... );
156 }
157
158 public:
159 template <typename... Gs>
160 Generators(Gs &&... moreGenerators) {
161 m_generators.reserve(sizeof...(Gs));
162 add_generators(CATCH_FORWARD(moreGenerators)...);
163 }
164
165 T const& get() const override {
166 return m_generators[m_current].get();
167 }
168
169 bool next() override {
170 if (m_current >= m_generators.size()) {
171 return false;
172 }
173 const bool current_status = m_generators[m_current].next();
174 if (!current_status) {
175 ++m_current;
176 }
177 return m_current < m_generators.size();
178 }
179
180 bool isFinite() const override {
181 for (auto const& gen : m_generators) {
182 if (!gen.isFinite()) { return false; }
183 }
184 return true;
185 }
186 };

222 template<typename L>
223 auto generate( StringRef generatorName, SourceLineInfo const& lineInfo, L const& generatorExpression ) -> typename decltype(generatorExpression())::type {
224 using UnderlyingType = typename decltype(generatorExpression())::type;
225
226 IGeneratorTracker* tracker = acquireGeneratorTracker( generatorName, lineInfo );
227 // Creation of tracker is delayed after generator creation, so
228 // that constructing generator can fail without breaking everything.
229 if (!tracker) {
230 tracker = createGeneratorTracker(
231 generatorName,
232 lineInfo,
233 Catch::Detail::make_unique<Generators<UnderlyingType>>(
234 generatorExpression() ) );
235 }
236
237 auto const& generator = static_cast<IGenerator<UnderlyingType> const&>( *tracker->getGenerator() );
238 return generator.get();
239 }
