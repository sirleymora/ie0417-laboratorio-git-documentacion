90 void XmlReporter::assertionEnded( AssertionStats const& assertionStats ) {
91
92 AssertionResult const& result = assertionStats.assertionResult;
93
94 bool includeResults = m_config->includeSuccessfulResults() || !result.isOk();
95
96 if( includeResults || result.getResultType() == ResultWas::Warning ) {
97 // Print any info messages in <Info> tags.
98 for( auto const& msg : assertionStats.infoMessages ) {
99 if( msg.type == ResultWas::Info && includeResults ) {
100 auto t = m_xml.scopedElement( "Info" );
101 writeSourceInfo( msg.lineInfo );
102 t.writeText( msg.message );
103 } else if ( msg.type == ResultWas::Warning ) {
104 auto t = m_xml.scopedElement( "Warning" );
105 writeSourceInfo( msg.lineInfo );
106 t.writeText( msg.message );
107 }
108 }
109 }
110
111 // Drop out if result was successful but we're not printing them.
112 if ( !includeResults && result.getResultType() != ResultWas::Warning &&
113 result.getResultType() != ResultWas::ExplicitSkip ) {
114 return;
115 }
116
117 // Print the expression if there is one.
118 if( result.hasExpression() ) {
119 m_xml.startElement( "Expression" )
120 .writeAttribute( "success"_sr, result.succeeded() )
121 .writeAttribute( "type"_sr, result.getTestMacroName() );
122
123 writeSourceInfo( result.getSourceInfo() );
124
125 m_xml.scopedElement( "Original" )
126 .writeText( result.getExpression() );
127 m_xml.scopedElement( "Expanded" )
128 .writeText( result.getExpandedExpression() );
129 }
130
131 // And... Print a result applicable to each result type.
132 switch( result.getResultType() ) {
133 case ResultWas::ThrewException:
134 m_xml.startElement( "Exception" );
135 writeSourceInfo( result.getSourceInfo() );
136 m_xml.writeText( result.getMessage() );
137 m_xml.endElement();
138 break;
139 case ResultWas::FatalErrorCondition:
140 m_xml.startElement( "FatalErrorCondition" );
141 writeSourceInfo( result.getSourceInfo() );
142 m_xml.writeText( result.getMessage() );
143 m_xml.endElement();
144 break;
145 case ResultWas::Info:
146 m_xml.scopedElement( "Info" )
147 .writeText( result.getMessage() );
148 break;
149 case ResultWas::Warning:
150 // Warning will already have been written
151 break;
152 case ResultWas::ExplicitFailure:
153 m_xml.startElement( "Failure" );
154 writeSourceInfo( result.getSourceInfo() );
155 m_xml.writeText( result.getMessage() );
156 m_xml.endElement();
157 break;
158 case ResultWas::ExplicitSkip:
159 m_xml.startElement( "Skip" );
160 writeSourceInfo( result.getSourceInfo() );
161 m_xml.writeText( result.getMessage() );
162 m_xml.endElement();
163 break;
164 default:
165 break;
166 }
167
168 if( result.hasExpression() )
169 m_xml.endElement();
170 }

172 void XmlReporter::sectionEnded( SectionStats const& sectionStats ) {
173 StreamingReporterBase::sectionEnded( sectionStats );
174 if ( --m_sectionDepth > 0 ) {
175 {
176 XmlWriter::ScopedElement e = m_xml.scopedElement( "OverallResults" );
177 e.writeAttribute( "successes"_sr, sectionStats.assertions.passed );
178 e.writeAttribute( "failures"_sr, sectionStats.assertions.failed );
179 e.writeAttribute( "expectedFailures"_sr, sectionStats.assertions.failedButOk );
180 e.writeAttribute( "skipped"_sr, sectionStats.assertions.skipped > 0 );
181
182 if ( m_config->showDurations() == ShowDurations::Always )
183 e.writeAttribute( "durationInSeconds"_sr, sectionStats.durationInSeconds );
184 }
185 // Ends assertion tag
186 m_xml.endElement();
187 }
188 }

190 void XmlReporter::testCaseEnded( TestCaseStats const& testCaseStats ) {
191 StreamingReporterBase::testCaseEnded( testCaseStats );
192 XmlWriter::ScopedElement e = m_xml.scopedElement( "OverallResult" );
193 e.writeAttribute( "success"_sr, testCaseStats.totals.assertions.allOk() );
194 e.writeAttribute( "skips"_sr, testCaseStats.totals.assertions.skipped );
195
196 if ( m_config->showDurations() == ShowDurations::Always )
197 e.writeAttribute( "durationInSeconds"_sr, m_testCaseTimer.getElapsedSeconds() );
198 if( !testCaseStats.stdOut.empty() )
199 m_xml.scopedElement( "StdOut" ).writeText( trim( StringRef(testCaseStats.stdOut) ), XmlFormatting::Newline );
200 if( !testCaseStats.stdErr.empty() )
201 m_xml.scopedElement( "StdErr" ).writeText( trim( StringRef(testCaseStats.stdErr) ), XmlFormatting::Newline );
202
203 m_xml.endElement();
204 }

235 void XmlReporter::benchmarkEnded(BenchmarkStats<> const& benchmarkStats) {
236 m_xml.scopedElement("mean")
237 .writeAttribute("value"_sr, benchmarkStats.mean.point.count())
238 .writeAttribute("lowerBound"_sr, benchmarkStats.mean.lower_bound.count())
239 .writeAttribute("upperBound"_sr, benchmarkStats.mean.upper_bound.count())
240 .writeAttribute("ci"_sr, benchmarkStats.mean.confidence_interval);
241 m_xml.scopedElement("standardDeviation")
242 .writeAttribute("value"_sr, benchmarkStats.standardDeviation.point.count())
243 .writeAttribute("lowerBound"_sr, benchmarkStats.standardDeviation.lower_bound.count())
244 .writeAttribute("upperBound"_sr, benchmarkStats.standardDeviation.upper_bound.count())
245 .writeAttribute("ci"_sr, benchmarkStats.standardDeviation.confidence_interval);
246 m_xml.scopedElement("outliers")
247 .writeAttribute("variance"_sr, benchmarkStats.outlierVariance)
248 .writeAttribute("lowMild"_sr, benchmarkStats.outliers.low_mild)
249 .writeAttribute("lowSevere"_sr, benchmarkStats.outliers.low_severe)
250 .writeAttribute("highMild"_sr, benchmarkStats.outliers.high_mild)
251 .writeAttribute("highSevere"_sr, benchmarkStats.outliers.high_severe);

287 void XmlReporter::listTests(std::vector<TestCaseHandle> const& tests) {
288 auto outerTag = m_xml.scopedElement("MatchingTests");
289 for (auto const& test : tests) {
290 auto innerTag = m_xml.scopedElement("TestCase");
291 auto const& testInfo = test.getTestCaseInfo();
292 m_xml.startElement("Name", XmlFormatting::Indent)
293 .writeText(testInfo.name, XmlFormatting::None)
294 .endElement(XmlFormatting::Newline);
295 m_xml.startElement("ClassName", XmlFormatting::Indent)
296 .writeText(testInfo.className, XmlFormatting::None)
297 .endElement(XmlFormatting::Newline);
298 m_xml.startElement("Tags", XmlFormatting::Indent)
299 .writeText(testInfo.tagsAsString(), XmlFormatting::None)
300 .endElement(XmlFormatting::Newline);
301
302 auto sourceTag = m_xml.scopedElement("SourceInfo");
303 m_xml.startElement("File", XmlFormatting::Indent)
304 .writeText(testInfo.lineInfo.file, XmlFormatting::None)
305 .endElement(XmlFormatting::Newline);
306 m_xml.startElement("Line", XmlFormatting::Indent)
307 .writeText(std::to_string(testInfo.lineInfo.line), XmlFormatting::None)
308 .endElement(XmlFormatting::Newline);
309 }
