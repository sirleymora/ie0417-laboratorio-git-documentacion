27 std::string getCurrentTimestamp() {
28 time_t rawtime;
29 std::time(&rawtime);
30
31 std::tm timeInfo = {};
32#if defined (_MSC_VER) || defined (__MINGW32__)
33 gmtime_s(&timeInfo, &rawtime);
34#elif defined (CATCH_PLATFORM_PLAYSTATION)
35 gmtime_s(&rawtime, &timeInfo);
36#elif defined (__IAR_SYSTEMS_ICC__)
37 timeInfo = *std::gmtime(&rawtime);
38#else
39 gmtime_r(&rawtime, &timeInfo);
40#endif
41
42 auto const timeStampSize = sizeof("2017-01-16T17:06:45Z");
43 char timeStamp[timeStampSize];
44 const char * const fmt = "%Y-%m-%dT%H:%M:%SZ";
45
46 std::strftime(timeStamp, timeStampSize, fmt, &timeInfo);
47
48 return std::string(timeStamp, timeStampSize - 1);
49 }

51 std::string fileNameTag(std::vector<Tag> const& tags) {
52 auto it = std::find_if(begin(tags),
53 end(tags),
54 [] (Tag const& tag) {
55 return tag.original.size() > 0
56 && tag.original[0] == '#'; });
57 if (it != tags.end()) {
58 return static_cast<std::string>(
59 it->original.substr(1, it->original.size() - 1)
60 );
61 }
62 return std::string();
63 }

75 static void normalizeNamespaceMarkers(std::string& str) {
76 std::size_t pos = str.find( "::" );
77 while ( pos != std::string::npos ) {
78 str.replace( pos, 2, "." );
79 pos += 1;
80 pos = str.find( "::", pos );
81 }

131 void JunitReporter::writeRun( TestRunNode const& testRunNode, double suiteTime ) {
132 XmlWriter::ScopedElement e = xml.scopedElement( "testsuite" );
133
134 TestRunStats const& stats = testRunNode.value;
135 xml.writeAttribute( "name"_sr, stats.runInfo.name );
136 xml.writeAttribute( "errors"_sr, unexpectedExceptions );
137 xml.writeAttribute( "failures"_sr, stats.totals.assertions.failed-unexpectedExceptions );
138 xml.writeAttribute( "skipped"_sr, stats.totals.assertions.skipped );
139 xml.writeAttribute( "tests"_sr, stats.totals.assertions.total() );
140 xml.writeAttribute( "hostname"_sr, "tbd"_sr ); // !TBD
141 if( m_config->showDurations() == ShowDurations::Never )
142 xml.writeAttribute( "time"_sr, ""_sr );
143 else
144 xml.writeAttribute( "time"_sr, formatDuration( suiteTime ) );
145 xml.writeAttribute( "timestamp"_sr, getCurrentTimestamp() );
146
147 // Write properties
148 {
149 auto properties = xml.scopedElement("properties");
150 xml.scopedElement("property")
151 .writeAttribute("name"_sr, "random-seed"_sr)
152 .writeAttribute("value"_sr, m_config->rngSeed());
153 if (m_config->testSpec().hasFilters()) {
154 xml.scopedElement("property")
155 .writeAttribute("name"_sr, "filters"_sr)
156 .writeAttribute("value"_sr, m_config->testSpec());
157 }
158 }
159
160 // Write test cases
161 for( auto const& child : testRunNode.children )
162 writeTestCase( *child );
163
164 xml.scopedElement( "system-out" ).writeText( trim( stdOutForSuite ), XmlFormatting::Newline );
165 xml.scopedElement( "system-err" ).writeText( trim( stdErrForSuite ), XmlFormatting::Newline );
166 }

168 void JunitReporter::writeTestCase( TestCaseNode const& testCaseNode ) {
169 TestCaseStats const& stats = testCaseNode.value;
170
171 // All test cases have exactly one section - which represents the
172 // test case itself. That section may have 0-n nested sections
173 assert( testCaseNode.children.size() == 1 );
174 SectionNode const& rootSection = *testCaseNode.children.front();
175
176 std::string className =
177 static_cast<std::string>( stats.testInfo->className );
178
179 if( className.empty() ) {
180 className = fileNameTag(stats.testInfo->tags);
181 if ( className.empty() ) {
182 className = "global";
183 }
184 }
185
186 if ( !m_config->name().empty() )
187 className = static_cast<std::string>(m_config->name()) + '.' + className;
188
189 normalizeNamespaceMarkers(className);
190
191 writeSection( className, "", rootSection, stats.testInfo->okToFail() );
192 }

194 void JunitReporter::writeSection( std::string const& className,
195 std::string const& rootName,
196 SectionNode const& sectionNode,
197 bool testOkToFail) {
198 std::string name = trim( sectionNode.stats.sectionInfo.name );
199 if( !rootName.empty() )
200 name = rootName + '/' + name;
201
202 if ( sectionNode.stats.assertions.total() > 0
203 || !sectionNode.stdOut.empty()
204 || !sectionNode.stdErr.empty() ) {
205 XmlWriter::ScopedElement e = xml.scopedElement( "testcase" );
206 if( className.empty() ) {
207 xml.writeAttribute( "classname"_sr, name );
208 xml.writeAttribute( "name"_sr, "root"_sr );
209 }
210 else {
211 xml.writeAttribute( "classname"_sr, className );
212 xml.writeAttribute( "name"_sr, name );
213 }
214 xml.writeAttribute( "time"_sr, formatDuration( sectionNode.stats.durationInSeconds ) );
215 // This is not ideal, but it should be enough to mimic gtest's
216 // junit output.
217 // Ideally the JUnit reporter would also handle `skipTest`
218 // events and write those out appropriately.
219 xml.writeAttribute( "status"_sr, "run"_sr );
220
221 if (sectionNode.stats.assertions.failedButOk) {
222 xml.scopedElement("skipped")
223 .writeAttribute("message", "TEST_CASE tagged with !mayfail");
224 }
225
226 writeAssertions( sectionNode );
227
228
229 if( !sectionNode.stdOut.empty() )
230 xml.scopedElement( "system-out" ).writeText( trim( sectionNode.stdOut ), XmlFormatting::Newline );
231 if( !sectionNode.stdErr.empty() )
232 xml.scopedElement( "system-err" ).writeText( trim( sectionNode.stdErr ), XmlFormatting::Newline );
233 }
234 for( auto const& childNode : sectionNode.childSections )
235 if( className.empty() )
236 writeSection( name, "", *childNode, testOkToFail );
237 else
238 writeSection( className, name, *childNode, testOkToFail );
239 }

241 void JunitReporter::writeAssertions( SectionNode const& sectionNode ) {
242 for (auto const& assertionOrBenchmark : sectionNode.assertionsAndBenchmarks) {
243 if (assertionOrBenchmark.isAssertion()) {
244 // JUnit XML format supports only 1 error/failure/skip
245 // assertion elements per test case
246 if (writeAssertion(assertionOrBenchmark.asAssertion())) {
247 break;
248 }
249 }
250 }

253 bool JunitReporter::writeAssertion( AssertionStats const& stats ) {
254 AssertionResult const& result = stats.assertionResult;
255 if ( result.isOk() &&
256 result.getResultType() != ResultWas::ExplicitSkip ) {
257 return false;
258 }
259 std::string elementName;
260 switch ( result.getResultType() ) {
261 case ResultWas::ThrewException:
262 case ResultWas::FatalErrorCondition:
263 elementName = "error";
264 break;
265 case ResultWas::ExplicitFailure:
266 case ResultWas::ExpressionFailed:
267 case ResultWas::DidntThrowException:
268 elementName = "failure";
269 break;
270 case ResultWas::ExplicitSkip:
271 elementName = "skipped";
272 break;
273 // We should never see these here:
274 case ResultWas::Info:
275 case ResultWas::Warning:
276 case ResultWas::Ok:
277 case ResultWas::Unknown:
278 case ResultWas::FailureBit:
279 case ResultWas::Exception:
280 elementName = "internalError";
281 break;
282 }
283
284 XmlWriter::ScopedElement e = xml.scopedElement( elementName );
285
286 xml.writeAttribute( "message"_sr, result.getExpression() );
287 xml.writeAttribute( "type"_sr, result.getTestMacroName() );
288
289 ReusableStringStream rss;
290 if ( result.getResultType() == ResultWas::ExplicitSkip ) {
291 rss << "SKIPPED\n";
292 } else {
293 rss << "FAILED:\n";
294 if ( result.hasExpression() ) {
295 rss << " ";
296 rss << result.getExpressionInMacro();
297 rss << '\n';
298 }
299 if ( result.hasExpandedExpression() ) {
300 rss << "with expansion:\n";
301 rss << TextFlow::Column( result.getExpandedExpression() )
302 .indent( 2 )
303 << '\n';
304 }
305 }
306
307 if ( result.hasMessage() ) { rss << result.getMessage() << '\n'; }
308 for ( auto const& msg : stats.infoMessages ) {
309 if ( msg.type == ResultWas::Info ) { rss << msg.message << '\n'; }
310 }
311
312 rss << "at " << result.getSourceInfo();
313 xml.writeText( rss.str(), XmlFormatting::Newline );
314 return true;
315 }
