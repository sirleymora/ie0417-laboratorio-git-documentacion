24 struct Endianness {
25 enum Arch : uint8_t {
26 Big,
27 Little
28 };
29
30 static Arch which() {
31 int one = 1;
32 // If the lowest byte we read is non-zero, we can assume
33 // that little endian format is used.
34 auto value = *reinterpret_cast<char*>(&one);
35 return value ? Little : Big;
36 }
37 };

39 template<typename T>
40 std::string fpToString(T value, int precision) {
41 if (Catch::isnan(value)) {
42 return "nan";
43 }
44
45 ReusableStringStream rss;
46 rss << std::setprecision(precision)
47 << std::fixed
48 << value;
49 std::string d = rss.str();
50 std::size_t i = d.find_last_not_of('0');
51 if (i != std::string::npos && i != d.size() - 1) {
52 if (d[i] == '.')
53 i++;
54 d = d.substr(0, i + 1);
55 }
56 return d;
57 }
58 } // end unnamed namespace

66 std::string formatTimeT(std::time_t time) {
67#ifdef _MSC_VER
68 std::tm timeInfo = {};
69 const auto err = gmtime_s( &timeInfo, &time );
70 if ( err ) {
71 return "gmtime from provided timepoint has failed. This "
72 "happens e.g. with pre-1970 dates using Microsoft libc";
73 }
74#else
75 std::tm* timeInfo = std::gmtime( &time );
76#endif
77
78 auto const timeStampSize = sizeof( "2017-01-16T17:06:45Z" );
79 char timeStamp[timeStampSize];
80 const char* const fmt = "%Y-%m-%dT%H:%M:%SZ";
81
82#ifdef _MSC_VER
83 std::strftime( timeStamp, timeStampSize, fmt, &timeInfo );
84#else
85 std::strftime( timeStamp, timeStampSize, fmt, timeInfo );
86#endif
87 return std::string( timeStamp, timeStampSize - 1 );
88 }

90 std::string convertIntoString(StringRef string, bool escapeInvisibles) {
91 std::string ret;
92 // This is enough for the "don't escape invisibles" case, and a good
93 // lower bound on the "escape invisibles" case.
94 ret.reserve( string.size() + 2 );
95
96 if ( !escapeInvisibles ) {
97 ret += '"';
98 ret += string;
99 ret += '"';
100 return ret;
101 }
102
103 size_t last_start = 0;
104 auto write_to = [&]( size_t idx ) {
105 if ( last_start < idx ) {
106 ret += string.substr( last_start, idx - last_start );
107 }
108 last_start = idx + 1;
109 };
110
111 ret += '"';
112 for ( size_t i = 0; i < string.size(); ++i ) {
113 const char c = string[i];
114 if ( c == '\r' || c == '\n' || c == '\t' || c == '\f' ) {
115 write_to( i );
116 if ( c == '\r' ) { ret.append( "\\r" ); }
117 if ( c == '\n' ) { ret.append( "\\n" ); }
118 if ( c == '\t' ) { ret.append( "\\t" ); }
119 if ( c == '\f' ) { ret.append( "\\f" ); }
120 }
121 }
122 write_to( string.size() );
123 ret += '"';
124
125 return ret;
126 }

132 std::string rawMemoryToString( const void *object, std::size_t size ) {
133 // Reverse order for little endian architectures
134 int i = 0, end = static_cast<int>( size ), inc = 1;
135 if( Endianness::which() == Endianness::Little ) {
136 i = end-1;
137 end = inc = -1;
138 }
139
140 unsigned char const *bytes = static_cast<unsigned char const *>(object);
141 ReusableStringStream rss;
142 rss << "0x" << std::setfill('0') << std::hex;
143 for( ; i != end; i += inc )
144 rss << std::setw(2) << static_cast<unsigned>(bytes[i]);
145 return rss.str();
146 }

258std::string StringMaker<signed char>::convert(signed char value) {
259 if (value == '\r') {
260 return "'\\r'";
261 } else if (value == '\f') {
262 return "'\\f'";
263 } else if (value == '\n') {
264 return "'\\n'";
265 } else if (value == '\t') {
266 return "'\\t'";
267 } else if ('\0' <= value && value < ' ') {
268 return ::Catch::Detail::stringify(static_cast<unsigned int>(value));
269 } else {
270 char chstr[] = "' '";
271 chstr[1] = value;
272 return chstr;
273 }
