21 template <typename T>
22 class TakeGenerator final : public IGenerator<T> {
23 GeneratorWrapper<T> m_generator;
24 size_t m_returned = 0;
25 size_t m_target;
26
27 void skipToNthElementImpl( std::size_t n ) override {
28 if ( n >= m_target ) {
29 Detail::throw_generator_exception(
30 "Could not jump to Nth element: not enough elements" );
31 }
32
33 m_generator.skipToNthElement( n );
34 m_returned = n;
35 }
36
37 public:
38 TakeGenerator(size_t target, GeneratorWrapper<T>&& generator):
39 m_generator(CATCH_MOVE(generator)),
40 m_target(target)
41 {
42 assert(target != 0 && "Empty generators are not allowed");
43 }
44 T const& get() const override {
45 return m_generator.get();
46 }
47 bool next() override {
48 ++m_returned;
49 if (m_returned >= m_target) {
50 return false;
51 }
52
53 const auto success = m_generator.next();
54 // If the underlying generator does not contain enough values
55 // then we cut short as well
56 if (!success) {
57 m_returned = m_target;
58 }
59 return success;
60 }
61
62 bool isFinite() const override { return true; }
63 };

66 GeneratorWrapper<T> take(size_t target, GeneratorWrapper<T>&& generator) {
67 return GeneratorWrapper<T>(Catch::Detail::make_unique<TakeGenerator<T>>(target, CATCH_MOVE(generator)));
68 }

71 template <typename T, typename Predicate>
72 class FilterGenerator final : public IGenerator<T> {
73 GeneratorWrapper<T> m_generator;
74 Predicate m_predicate;
75 static_assert(!std::is_reference<Predicate>::value, "This would most likely result in a dangling reference");
76 public:
77 template <typename P>
78 FilterGenerator(P&& pred, GeneratorWrapper<T>&& generator):
79 m_generator(CATCH_MOVE(generator)),
80 m_predicate(CATCH_FORWARD(pred))
81 {
82 if (!m_predicate(m_generator.get())) {
83 // It might happen that there are no values that pass the
84 // filter. In that case we throw an exception.
85 auto has_initial_value = next();
86 if (!has_initial_value) {
87 Detail::throw_generator_exception("No valid value found in filtered generator");
88 }
89 }
90 }
91
92 T const& get() const override {
93 return m_generator.get();
94 }
95
96 bool next() override {
97 bool success = m_generator.next();
98 if (!success) {
99 return false;
100 }
101 while (!m_predicate(m_generator.get()) && (success = m_generator.next()) == true);
102 return success;
103 }
104
105 bool isFinite() const override { return m_generator.isFinite(); }
106 };

114 template <typename T>
115 class RepeatGenerator final : public IGenerator<T> {
116 static_assert(!std::is_same<T, bool>::value,
117 "RepeatGenerator currently does not support bools"
118 "because of std::vector<bool> specialization");
119 GeneratorWrapper<T> m_generator;
120 mutable std::vector<T> m_returned;
121 size_t m_target_repeats;
122 size_t m_current_repeat = 0;
123 size_t m_repeat_index = 0;
124 public:
125 RepeatGenerator(size_t repeats, GeneratorWrapper<T>&& generator):
126 m_generator(CATCH_MOVE(generator)),
127 m_target_repeats(repeats)
128 {
129 assert(m_target_repeats > 0 && "Repeat generator must repeat at least once");
130 if (!m_generator.isFinite()) {
131 Detail::throw_generator_exception( "Cannot repeat infinite generator" );
132 }
133 }
134
135 T const& get() const override {
136 if (m_current_repeat == 0) {
137 m_returned.push_back(m_generator.get());
138 return m_returned.back();
139 }
140 return m_returned[m_repeat_index];
141 }
142
143 bool next() override {
144 // There are 2 basic cases:
145 // 1) We are still reading the generator
146 // 2) We are reading our own cache
147
148 // In the first case, we need to poke the underlying generator.
149 // If it happily moves, we are left in that state, otherwise it is time to start reading from our cache
150 if (m_current_repeat == 0) {
151 const auto success = m_generator.next();
152 if (!success) {
153 ++m_current_repeat;
154 }
155 return m_current_repeat < m_target_repeats;
156 }
157
158 // In the second case, we need to move indices forward and check that we haven't run up against the end
159 ++m_repeat_index;
160 if (m_repeat_index == m_returned.size()) {
161 m_repeat_index = 0;
162 ++m_current_repeat;
163 }
164 return m_current_repeat < m_target_repeats;
165 }
166
167 bool isFinite() const override { return m_generator.isFinite(); }
168 };

176 class MapGenerator final : public IGenerator<T> {
177 // TBD: provide static assert for mapping function, for friendly error message
178 GeneratorWrapper<U> m_generator;
179 Func m_function;
180 // To avoid returning dangling reference, we have to save the values
181 mutable Optional<T> m_cache;
182
183 void skipToNthElementImpl( std::size_t n ) override {
184 m_generator.skipToNthElement( n );
185 m_cache.reset();
186 }
187
188 public:
189 template <typename F2 = Func>
190 MapGenerator(F2&& function, GeneratorWrapper<U>&& generator) :
191 m_generator(CATCH_MOVE(generator)),
192 m_function(CATCH_FORWARD(function))
193 {}
194
195 T const& get() const override {
196 if ( !m_cache ) { m_cache = m_function( m_generator.get() ); }
197 return *m_cache;
198 }
199 bool next() override {
200 m_cache.reset();
201 return m_generator.next();
202 }
203
204 bool isFinite() const override { return m_generator.isFinite(); }
205 };

221 template <typename T>
222 class ChunkGenerator final : public IGenerator<std::vector<T>> {
223 std::vector<T> m_chunk;
224 size_t m_chunk_size;
225 GeneratorWrapper<T> m_generator;
226 public:
227 ChunkGenerator(size_t size, GeneratorWrapper<T> generator) :
228 m_chunk_size(size), m_generator(CATCH_MOVE(generator))
229 {
230 m_chunk.reserve(m_chunk_size);
231 if (m_chunk_size != 0) {
232 m_chunk.push_back(m_generator.get());
233 for (size_t i = 1; i < m_chunk_size; ++i) {
234 if (!m_generator.next()) {
235 Detail::throw_generator_exception("Not enough values to initialize the first chunk");
236 }
237 m_chunk.push_back(m_generator.get());
238 }
239 }
240 }
241 std::vector<T> const& get() const override {
242 return m_chunk;
243 }
244 bool next() override {
245 m_chunk.clear();
246 for (size_t idx = 0; idx < m_chunk_size; ++idx) {
247 if (!m_generator.next()) {
248 return false;
249 }
250 m_chunk.push_back(m_generator.get());
251 }
252 return true;
253 }
254
255 bool isFinite() const override { return m_generator.isFinite(); }
256 };

265 template <typename T>
266 class ConcatGenerator final : public IGenerator<T> {
267 std::vector<GeneratorWrapper<T>> m_generators;
268 size_t m_current_generator = 0;
269
270 void InsertGenerators( GeneratorWrapper<T>&& gen ) {
271 m_generators.push_back( CATCH_MOVE( gen ) );
272 }
273
274 template <typename... Generators>
275 void InsertGenerators( GeneratorWrapper<T>&& gen, Generators&&... gens ) {
276 m_generators.push_back( CATCH_MOVE( gen ) );
277 InsertGenerators( CATCH_MOVE( gens )... );
278 }
279
280 public:
281 template <typename... Generators>
282 ConcatGenerator( Generators&&... generators ) {
283 InsertGenerators( CATCH_MOVE( generators )... );
284 }
285
286 T const& get() const override {
287 return m_generators[m_current_generator].get();
288 }
289 bool next() override {
290 const bool success = m_generators[m_current_generator].next();
291 if ( success ) { return true; }
292
293 // If current generator is used up, we have to move to the next one
294 ++m_current_generator;
295 return m_current_generator < m_generators.size();
296 }
297
298 bool isFinite() const override {
299 for ( auto const& gen : m_generators ) {
300 if ( !gen.isFinite() ) { return false; }
301 }
302 return true;
303 }
304 };
