37 static Catch::StringRef makeEscapeStringRef( char c ) {
38 if ( c == '"' ) {
39 return "\\\""_sr;
40 } else if ( c == '\\' ) {
41 return "\\\\"_sr;
42 } else if ( c == '\b' ) {
43 return "\\b"_sr;
44 } else if ( c == '\f' ) {
45 return "\\f"_sr;
46 } else if ( c == '\n' ) {
47 return "\\n"_sr;
48 } else if ( c == '\r' ) {
49 return "\\r"_sr;
50 } else if ( c == '\t' ) {
51 return "\\t"_sr;
52 }
53 Catch::Detail::Unreachable();

176 void JsonValueWriter::writeFloatingPoint( T value ) {
177 if ( Catch::isnan( value ) ) {
178 writeImpl( "NaN"_sr, false );
179 } else if ( std::isinf( value ) ) {
180 writeImpl( value < 0 ? "-Infinity"_sr : "Infinity"_sr, false );
181 } else {
182 m_sstream << value;
183 writeImpl( m_sstream.str(), false );
184 }

187 void JsonValueWriter::writeImpl( Catch::StringRef value, bool quote ) {
188 // Escaping only makes sense for actual strings, which are passed
189 // with `quote == true`. Non-quoted values are things like bools
190 // and numbers, which cannot create inputs that need escaping.
191 if ( !quote ) {
192 m_os << value;
193 } else {
194 m_os << '"';
195 size_t current_start = 0;
196 for ( size_t i = 0; i < value.size(); ++i ) {
197 if ( needsEscape( value[i] ) ) {
198 if ( current_start < i ) {
199 m_os
200 << value.substr( current_start, i - current_start );
201 }
202 m_os << makeEscapeStringRef( value[i] );
203 current_start = i + 1;
204 }
205 }
206 if ( current_start < value.size() ) {
207 m_os << value.substr( current_start,
208 value.size() - current_start );
209 }
210 m_os << '"';
211 }
212 }
