52 bool StringEqualsMatcher::match( std::string const& source ) const {
53 if (m_caseSensitivity == CaseSensitive::Yes) {

63 std::string comparator, CaseSensitive caseSensitivity ):
64 StringMatcherBase( CATCH_MOVE( comparator ), "contains"_sr, caseSensitivity ) {}
65
66 bool StringContainsMatcher::match( std::string const& source ) const {
67 if ( m_caseSensitivity == CaseSensitive::Yes ) {
68 return contains( source, m_target );
69 }
70 if ( source.size() < m_target.size() ) { return false; }
71 StringRef as_ref( source );
72 // The worst case of this is O(m*n), which is terrible, BUT:
73 // * The average case is much better, the worst case only happens rarely
74 // * We can implement BMH/other better searchers later if it matters
75 Catch::Detail::CaseInsensitiveEqualTo eq;
76 for (size_t i = 0; i < source.size(); ++i) {
77 const auto substr = as_ref.substr( i, m_target.size() );
78 bool found = eq( substr, m_target );
79 if ( found ) { return true; }
80 }
81 return false;
82 }

109 Catch::Detail::CaseInsensitiveEqualTo eq;

116 RegexMatcher::RegexMatcher(std::string regex, CaseSensitive caseSensitivity): m_regex(CATCH_MOVE(regex)), m_caseSensitivity(caseSensitivity) {}
117
118 bool RegexMatcher::match(std::string const& matchee) const {
119 auto flags = std::regex::ECMAScript; // ECMAScript is the default syntax option anyway
120 if (m_caseSensitivity == CaseSensitive::No) {
121 flags |= std::regex::icase;
122 }
123 auto reg = std::regex(m_regex, flags);
124 return std::regex_match(matchee, reg);
125 }

127 std::string RegexMatcher::describe() const {
128 return "matches " + ::Catch::Detail::stringify(m_regex) + ((m_caseSensitivity == CaseSensitive::Yes)? " case sensitively" : " case insensitively");
129 }
