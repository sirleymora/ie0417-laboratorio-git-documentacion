38 struct GeneratorTracker final : TestCaseTracking::TrackerBase,
39 IGeneratorTracker {
40 GeneratorBasePtr m_generator;
41 // Filtered generator has moved to specific index due to
42 // a filter, it needs special handling of `countedNext()`
43 bool m_isFiltered = false;
44
45 GeneratorTracker(
46 TestCaseTracking::NameAndLocation&& nameAndLocation,
47 TrackerContext& ctx,
48 ITracker* parent,
49 GeneratorBasePtr&& generator ):
50 TrackerBase( CATCH_MOVE( nameAndLocation ), ctx, parent ),
51 m_generator( CATCH_MOVE( generator ) ) {
52 assert( m_generator &&
53 "Cannot create tracker without generator" );
54
55 // Handle potential filter and move forward here...
56 // Old style filters do not affect generators at all
57 if (m_newStyleFilters && m_allTrackerDepth < m_filterRef->size()) {
58 auto const& filter =
59 ( *m_filterRef )[m_allTrackerDepth];
60 // Generator cannot be un-entered the way a section
61 // can be, so the tracker has to throw for a wrong
62 // filter to stop the execution flow.
63 if (filter.type == PathFilter::For::Section) {
64 // We want the semantics of `SKIP()`, but we inline it
65 // to avoid issues with conditionally prefixed macros
66 INTERNAL_CATCH_MSG(
67 "SKIP",
68 Catch::ResultWas::ExplicitSkip,
69 Catch::ResultDisposition::Normal,
70 "" );
71 Catch::Detail::Unreachable();
72 }
73 // '*' is the wildcard for "all elements in generator"
74 // used for filtering sections below the generator, but
75 // not the generator itself.
76 if ( filter.filter != "*" ) {
77 m_isFiltered = true;
78 // TBD: We assume that the filter was validated as
79 // number during parsing. We should pass it
80 // as number from the CLI parser.
81 size_t targetIndex = std::stoul( filter.filter );
82 m_generator->skipToNthElement( targetIndex );
83 }
84 }
85 }

87 static GeneratorTracker*
88 acquire( TrackerContext& ctx,
89 TestCaseTracking::NameAndLocationRef const&
90 nameAndLocation ) {
91 GeneratorTracker* tracker;
92
93 ITracker& currentTracker = ctx.currentTracker();
94 // Under specific circumstances, the generator we want
95 // to acquire is also the current tracker. If this is
96 // the case, we have to avoid looking through current
97 // tracker's children, and instead return the current
98 // tracker.
99 // A case where this check is important is e.g.
100 // for (int i = 0; i < 5; ++i) {
101 // int n = GENERATE(1, 2);
102 // }
103 //
104 // without it, the code above creates 5 nested generators.
105 if ( currentTracker.nameAndLocation() == nameAndLocation ) {
106 auto thisTracker = currentTracker.parent()->findChild(
107 nameAndLocation );
108 assert( thisTracker );
109 assert( thisTracker->isGeneratorTracker() );
110 tracker = static_cast<GeneratorTracker*>( thisTracker );
111 } else if ( ITracker* childTracker =
112 currentTracker.findChild(
113 nameAndLocation ) ) {
114 assert( childTracker );
115 assert( childTracker->isGeneratorTracker() );
116 tracker =
117 static_cast<GeneratorTracker*>( childTracker );
118 } else {
119 return nullptr;
120 }
121
122 if ( !tracker->isComplete() ) { tracker->open(); }
123
124 return tracker;
125 }

127 // TrackerBase interface
128 bool isGeneratorTracker() const override { return true; }
129 void close() override {
130 TrackerBase::close();
131 // If a generator has a child (it is followed by a section)
132 // and none of its children have started, then we must wait
133 // until later to start consuming its values.
134 // This catches cases where `GENERATE` is placed between two
135 // `SECTION`s.
136 // **The check for m_children.empty cannot be removed**.
137 // doing so would break `GENERATE` _not_ followed by
138 // `SECTION`s.
139 const bool should_wait_for_child = [&]() {
140 // No children -> nobody to wait for
141 if ( m_children.empty() ) { return false; }
142 // If at least one child started executing, don't wait
143 if ( std::find_if(
144 m_children.begin(),
145 m_children.end(),
146 []( TestCaseTracking::ITrackerPtr const&
147 tracker ) {
148 return tracker->hasStarted();
149 } ) != m_children.end() ) {
150 return false;
151 }
152
153 // No children have started. We need to check if they
154 // _can_ start, and thus we should wait for them, or
155 // they cannot start (due to filters), and we shouldn't
156 // wait for them
157
158 // No filters left -> no restrictions on running sections
159 size_t childDepth = 1 + (m_newStyleFilters ? m_allTrackerDepth : m_sectionOnlyDepth);
160 if ( childDepth >= m_filterRef->size() ) {
161 return true;
162 }
163
164 // If we are using the new style filters, we need to check
165 // whether the successive filter is for section or a generator.
166 if ( m_newStyleFilters
167 && (*m_filterRef)[childDepth].type != PathFilter::For::Section ) {
168 return false;
169 }
170 // Look for any child section that could match the remaining filters
171 for ( auto const& child : m_children ) {
172 if ( child->isSectionTracker() &&
173 static_cast<SectionTracker const&>( *child )
174 .trimmedName() == StringRef((*m_filterRef)[childDepth].filter) ) {
175 return true;
176 }
177 }
178 return false;
179 }();
180
181 // This check is a bit tricky, because m_generator->next()
182 // has a side-effect, where it consumes generator's current
183 // value, but we do not want to invoke the side-effect if
184 // this generator is still waiting for any child to start.
185 assert( m_generator && "Tracker without generator" );
186 if ( should_wait_for_child
187 || ( m_runState == CompletedSuccessfully
188 && !m_isFiltered // filtered generators cannot meaningfully move forward, as they would get past the filter
189 && m_generator->countedNext() ) ) {
190 m_children.clear();
191 m_runState = Executing;
192 }
193 }

195 // IGeneratorTracker interface
196 auto getGenerator() const -> GeneratorBasePtr const& override {
197 return m_generator;
198 }
199
200 bool isFilteredImpl() const override { return m_isFiltered; }
201 };

206 // Assertions are owned by the thread that is executing them.
207 // This allows for lock-free progress in common cases where we
208 // do not need to send the assertion events to the reporter.
209 // This also implies that messages are owned by their respective
210 // threads, and should not be shared across different threads.
211 //
212 // This implies that various pieces of metadata referring to last
213 // assertion result/source location/message handling, etc
214 // should also be thread local. For now we just use naked globals
215 // below, in the future we will want to allocate piece of memory
216 // from heap, to avoid consuming too much thread-local storage.
217 //
218 // Note that we also don't want non-trivial the thread-local variables
219 // below be initialized for every thread, only for those that touch
220 // Catch2. To make this work with both GCC/Clang and MSVC, we have to
221 // make them thread-local magic statics. (Class-level statics have the
222 // desired semantics on GCC, but not on MSVC).
223
224 // This is used for the "if" part of CHECKED_IF/CHECKED_ELSE
225 static CATCH_INTERNAL_THREAD_LOCAL bool g_lastAssertionPassed = false;
226
227 // This is the source location for last encountered macro. It is
228 // used to provide the users with more precise location of error
229 // when an unexpected exception/fatal error happens.
230 static CATCH_INTERNAL_THREAD_LOCAL SourceLineInfo
231 g_lastKnownLineInfo( "DummyLocation", static_cast<size_t>( -1 ) );
232
233 // Should we clear message scopes before sending off the messages to
234 // reporter? Set in `assertionPassedFastPath` to avoid doing the full
235 // clear there for performance reasons.
236 static CATCH_INTERNAL_THREAD_LOCAL bool g_clearMessageScopes = false;

239 // Holds the data for both scoped and unscoped messages together,
240 // to avoid issues where their lifetimes start in wrong order,
241 // and then are destroyed in wrong order.
242 class MessageHolder {
243 // The actual message vector passed to the reporters
244 std::vector<MessageInfo> messages;
245 // IDs of messages from UNSCOPED_X macros, which we have to
246 // remove manually.
247 std::vector<unsigned int> unscoped_ids;
248
249 public:
250 // We do not need to special-case the unscoped messages when
251 // we only keep around the raw msg ids.
252 ~MessageHolder() = default;
253
254 void addUnscopedMessage( MessageInfo&& info ) {
255 repairUnscopedMessageInvariant();
256 unscoped_ids.push_back( info.sequence );
257 messages.push_back( CATCH_MOVE( info ) );
258 }
259
260 void addUnscopedMessage(MessageBuilder&& builder) {
261 MessageInfo info( CATCH_MOVE( builder.m_info ) );
262 info.message = builder.m_stream.str();
263 addUnscopedMessage( CATCH_MOVE( info ) );
264 }
265
266 void addScopedMessage(MessageInfo&& info) {
267 messages.push_back( CATCH_MOVE( info ) );
268 }
269
270 std::vector<MessageInfo> const& getMessages() const {
271 return messages;
272 }
273
274 void removeMessage( unsigned int messageId ) {
275 // Note: On average, it would probably be better to look for
276 // the message backwards. However, we do not expect to have
277 // to deal with more messages than low single digits, so
278 // the improvement is tiny, and we would have to hand-write
279 // the loop to avoid terrible codegen of reverse iterators
280 // in debug mode.
281 auto iter =
282 std::find_if( messages.begin(),
283 messages.end(),
284 [messageId]( MessageInfo const& msg ) {
285 return msg.sequence == messageId;
286 } );
287 assert( iter != messages.end() &&
288 "Trying to remove non-existent message." );
289 messages.erase( iter );
290 }
291
292 void removeUnscopedMessages() {
293 for ( const auto messageId : unscoped_ids ) {
294 removeMessage( messageId );
295 }
296 unscoped_ids.clear();
297 g_clearMessageScopes = false;
298 }
299
300 void repairUnscopedMessageInvariant() {
301 if ( g_clearMessageScopes ) { removeUnscopedMessages(); }
302 g_clearMessageScopes = false;
303 }
304 };

361 Totals RunContext::runTest(TestCaseHandle const& testCase) {
362 updateTotalsFromAtomics();
363 const Totals prevTotals = m_totals;
364
365 auto const& testInfo = testCase.getTestCaseInfo();
366 m_reporter->testCaseStarting(testInfo);
367 testCase.prepareTestCase();
368 m_activeTestCase = &testCase;
369
370
371 ITracker& rootTracker = m_trackerContext.startRun();
372 assert(rootTracker.isSectionTracker());
373 rootTracker.setFilters( &m_config->getPathFilters(),
374 m_config->useNewFilterBehaviour() );
375
376 // We intentionally only seed the internal RNG once per test case,
377 // before it is first invoked. The reason for that is a complex
378 // interplay of generator/section implementation details and the
379 // Random*Generator types.
380 //
381 // The issue boils down to us needing to seed the Random*Generators
382 // with different seed each, so that they return different sequences
383 // of random numbers. We do this by giving them a number from the
384 // shared RNG instance as their seed.
385 //
386 // However, this runs into an issue if the reseeding happens each
387 // time the test case is entered (as opposed to first time only),
388 // because multiple generators could get the same seed, e.g. in
389 // ```cpp
390 // TEST_CASE() {
391 // auto i = GENERATE(take(10, random(0, 100));
392 // SECTION("A") {
393 // auto j = GENERATE(take(10, random(0, 100));
394 // }
395 // SECTION("B") {
396 // auto k = GENERATE(take(10, random(0, 100));
397 // }
398 // }
399 // ```
400 // `i` and `j` would properly return values from different sequences,
401 // but `i` and `k` would return the same sequence, because their seed
402 // would be the same.
403 // (The reason their seeds would be the same is that the generator
404 // for k would be initialized when the test case is entered the second
405 // time, after the shared RNG instance was reset to the same value
406 // it had when the generator for i was initialized.)
407 seedRng( *m_config );
408
409 uint64_t testRuns = 0;
410 std::string redirectedCout;
411 std::string redirectedCerr;
412 do {
413 m_trackerContext.startCycle();
414 m_testCaseTracker = &SectionTracker::acquire(m_trackerContext, TestCaseTracking::NameAndLocationRef(testInfo.name, testInfo.lineInfo));
415
416 m_reporter->testCasePartialStarting(testInfo, testRuns);
417
418 updateTotalsFromAtomics();
419 const auto beforeRunTotals = m_totals;
420 runCurrentTest();
421 std::string oneRunCout = m_outputRedirect->getStdout();
422 std::string oneRunCerr = m_outputRedirect->getStderr();
423 m_outputRedirect->clearBuffers();
424 redirectedCout += oneRunCout;
425 redirectedCerr += oneRunCerr;
426
427 updateTotalsFromAtomics();
428 const auto singleRunTotals = m_totals.delta(beforeRunTotals);
429 auto statsForOneRun = TestCaseStats(testInfo, singleRunTotals, CATCH_MOVE(oneRunCout), CATCH_MOVE(oneRunCerr), aborting());
430 m_reporter->testCasePartialEnded(statsForOneRun, testRuns);
431
432 ++testRuns;
433 } while (!m_testCaseTracker->isSuccessfullyCompleted() && !aborting());
434
435 Totals deltaTotals = m_totals.delta(prevTotals);
436 if (testInfo.expectedToFail() && deltaTotals.testCases.passed > 0) {
437 deltaTotals.assertions.failed++;
438 deltaTotals.testCases.passed--;
439 deltaTotals.testCases.failed++;
440 }
441 m_totals.testCases += deltaTotals.testCases;
442 testCase.tearDownTestCase();
443 m_reporter->testCaseEnded(TestCaseStats(testInfo,
444 deltaTotals,
445 CATCH_MOVE(redirectedCout),
446 CATCH_MOVE(redirectedCerr),
447 aborting()));
448
449 m_activeTestCase = nullptr;
450 m_testCaseTracker = nullptr;
451
452 return deltaTotals;
453 }

456 void RunContext::assertionEnded(AssertionResult&& result) {
457 Detail::g_lastKnownLineInfo = result.m_info.lineInfo;
458 if (result.getResultType() == ResultWas::Ok) {
459 m_atomicAssertionCount.passed++;
460 Detail::g_lastAssertionPassed = true;
461 } else if (result.getResultType() == ResultWas::ExplicitSkip) {
462 m_atomicAssertionCount.skipped++;
463 Detail::g_lastAssertionPassed = true;
464 } else if (!result.succeeded()) {
465 Detail::g_lastAssertionPassed = false;
466 if (result.isOk()) {}
467 else if( m_activeTestCase->getTestCaseInfo().okToFail() ) { // Read from a shared state established before the threads could start, this is fine
468 m_atomicAssertionCount.failedButOk++;
469 } else {
470 m_atomicAssertionCount.failed++;
471 }
472 } else {
473 Detail::g_lastAssertionPassed = true;
474 }
475
476 auto& msgHolder = Detail::g_messageHolder();
477 msgHolder.repairUnscopedMessageInvariant();
478
479 // From here, we are touching shared state and need mutex.
480 Detail::LockGuard lock( m_assertionMutex );
481 {
482 auto _ = scopedDeactivate( *m_outputRedirect );
483 updateTotalsFromAtomics();
484 m_reporter->assertionEnded( AssertionStats( result, msgHolder.getMessages(), m_totals ) );
485 }
486
487 if ( result.getResultType() != ResultWas::Warning ) {
488 msgHolder.removeUnscopedMessages();
489 }
490
491 // Reset working state. assertion info will be reset after
492 // populateReaction is run if it is needed
493 m_lastResult = CATCH_MOVE( result );
494 }

540 IGeneratorTracker* RunContext::createGeneratorTracker(
541 StringRef generatorName,
542 SourceLineInfo lineInfo,
543 Generators::GeneratorBasePtr&& generator ) {
544
545 auto nameAndLoc = TestCaseTracking::NameAndLocation( static_cast<std::string>( generatorName ), lineInfo );
546 auto& currentTracker = m_trackerContext.currentTracker();
547 assert(
548 currentTracker.nameAndLocation() != nameAndLoc &&
549 "Trying to create tracker for a generator that already has one" );
550
551 auto newTracker =
552 Catch::Detail::make_unique<Generators::GeneratorTracker>(
553 CATCH_MOVE( nameAndLoc ),
554 m_trackerContext,
555 &currentTracker,
556 CATCH_MOVE( generator ) );
557
558 // The warning shouldn't fire if the generator is infinite, **but** filtered down.
559 if ( m_config->warnAboutInfiniteGenerators() &&
560 !newTracker->m_generator->isFinite() &&
561 !newTracker->isFiltered() ) {
562 // We want the semantics of `FAIL()`, but we inline it
563 // to avoid issues with conditionally prefixed macros
564 INTERNAL_CATCH_MSG( "FAIL",
565 Catch::ResultWas::ExplicitFailure,
566 Catch::ResultDisposition::Normal,
567 "GENERATE() would run infinitely" );
568 }
569
570 auto returnPtr = newTracker.get();
571 currentTracker.addChild( CATCH_MOVE( newTracker ) );
572
573 returnPtr->open();
574 return returnPtr;
575 }

663 void RunContext::handleFatalErrorCondition( StringRef message ) {
664 // We lock only when touching the reporters directly, to avoid
665 // deadlocks when we call into other functions that also want
666 // to lock the mutex before touching reporters.
667 //
668 // This does mean that we allow other threads to run while handling
669 // a fatal error, but this is all a best effort attempt anyway.
670 {
671 Detail::LockGuard lock( m_assertionMutex );
672 // TODO: scoped deactivate here? Just give up and do best effort?
673 // the deactivation can break things further, OTOH so can the
674 // capture
675 auto _ = scopedDeactivate( *m_outputRedirect );
676
677 // First notify reporter that bad things happened
678 m_reporter->fatalErrorEncountered( message );
679 }
680
681 // Don't rebuild the result -- the stringification itself can cause more fatal errors
682 // Instead, fake a result data.
683 AssertionResultData tempResult( ResultWas::FatalErrorCondition, { false } );
684 tempResult.message = static_cast<std::string>(message);
685 AssertionResult result( makeDummyAssertionInfo(),
686 CATCH_MOVE( tempResult ) );
687
688 assertionEnded(CATCH_MOVE(result) );
689
690
691 // At this point we touch sections/test cases from this thread
692 // to try and end them. Technically that is not supported when
693 // using multiple threads, but the worst thing that can happen
694 // is that the process aborts harder :-D
695 Detail::LockGuard lock( m_assertionMutex );
696
697 // Best effort cleanup for sections that have not been destructed yet
698 // Since this is a fatal error, we have not had and won't have the opportunity to destruct them properly
699 while (!m_activeSections.empty()) {
700 auto const& nl = m_activeSections.back()->nameAndLocation();
701 SectionEndInfo endInfo{ SectionInfo(nl.location, nl.name), {}, 0.0 };
702 sectionEndedEarly(CATCH_MOVE(endInfo));
703 }
704 handleUnfinishedSections();
705
706 // Recreate section for test case (as we will lose the one that was in scope)
707 auto const& testCaseInfo = m_activeTestCase->getTestCaseInfo();
708 SectionInfo testCaseSection(testCaseInfo.lineInfo, testCaseInfo.name);
709
710 Counts assertions;
711 assertions.failed = 1;
712 SectionStats testCaseSectionStats(CATCH_MOVE(testCaseSection), assertions, 0, false);
713 m_reporter->sectionEnded( testCaseSectionStats );
714
715 auto const& testInfo = m_activeTestCase->getTestCaseInfo();
716
717 Totals deltaTotals;
718 deltaTotals.testCases.failed = 1;
719 deltaTotals.assertions.failed = 1;
720 m_reporter->testCaseEnded(TestCaseStats(testInfo,
721 deltaTotals,
722 std::string(),
723 std::string(),
724 false));
725 m_totals.testCases.failed++;
726 updateTotalsFromAtomics();
727 m_reporter->testRunEnded(TestRunStats(m_runInfo, m_totals, false));
728 }

751 void RunContext::runCurrentTest() {
752 auto const& testCaseInfo = m_activeTestCase->getTestCaseInfo();
753 SectionInfo testCaseSection(testCaseInfo.lineInfo, testCaseInfo.name);
754 m_reporter->sectionStarting(testCaseSection);
755 updateTotalsFromAtomics();
756 Counts prevAssertions = m_totals.assertions;
757 double duration = 0;
758 m_shouldReportUnexpected = true;
759 Detail::g_lastKnownLineInfo = testCaseInfo.lineInfo;
760
761 Timer timer;
762 CATCH_TRY {
763 {
764 auto _ = scopedActivate( *m_outputRedirect );
765 timer.start();
766 invokeActiveTestCase();
767 }
768 duration = timer.getElapsedSeconds();
769 } CATCH_CATCH_ANON (TestFailureException&) {
770 // This just means the test was aborted due to failure
771 } CATCH_CATCH_ANON (TestSkipException&) {
772 // This just means the test was explicitly skipped
773 } CATCH_CATCH_ALL {
774 // Under CATCH_CONFIG_FAST_COMPILE, unexpected exceptions under REQUIRE assertions
775 // are reported without translation at the point of origin.
776 if ( m_shouldReportUnexpected ) {
777 AssertionReaction dummyReaction;
778 handleUnexpectedInflightException( makeDummyAssertionInfo(),
779 translateActiveException(),
780 dummyReaction );
781 }
782 }
783 updateTotalsFromAtomics();
784 Counts assertions = m_totals.assertions - prevAssertions;
785 bool missingAssertions = testForMissingAssertions(assertions);
786
787 m_testCaseTracker->close();
788 handleUnfinishedSections();
789 auto& msgHolder = Detail::g_messageHolder();
790 msgHolder.removeUnscopedMessages();
791 assert( msgHolder.getMessages().empty() &&
792 "There should be no leftover messages after the test ends" );
793
794 SectionStats testCaseSectionStats(CATCH_MOVE(testCaseSection), assertions, duration, missingAssertions);
795 m_reporter->sectionEnded(testCaseSectionStats);
796 }

799 // We need to engage a handler for signals/structured exceptions
800 // before running the tests themselves, or the binary can crash
801 // without failed test being reported.
802 FatalConditionHandlerGuard _(&m_fatalConditionhandler);
803 // We keep having issue where some compilers warn about an unused
804 // variable, even though the type has non-trivial constructor and
805 // destructor. This is annoying and ugly, but it makes them stfu.
806 (void)_;
807
808 m_activeTestCase->invoke();
809 }
