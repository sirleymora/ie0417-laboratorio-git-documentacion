50 TestCaseProperties parseSpecialTag( StringRef tag ) {
51 if( !tag.empty() && tag[0] == '.' )
52 return TestCaseProperties::IsHidden;
53 else if( tag == "!throws"_sr )
54 return TestCaseProperties::Throws;
55 else if( tag == "!shouldfail"_sr )
56 return TestCaseProperties::ShouldFail;
57 else if( tag == "!mayfail"_sr )
58 return TestCaseProperties::MayFail;
59 else if( tag == "!nonportable"_sr )
60 return TestCaseProperties::NonPortable;
61 else if( tag == "!benchmark"_sr )
62 return TestCaseProperties::Benchmark | TestCaseProperties::IsHidden;
63 else
64 return TestCaseProperties::None;
65 }
66 bool isReservedTag( StringRef tag ) {
67 return parseSpecialTag( tag ) == TestCaseProperties::None
68 && tag.size() > 0
69 && !std::isalnum( static_cast<unsigned char>(tag[0]) );
70 }
71 void enforceNotReservedTag( StringRef tag, SourceLineInfo const& _lineInfo ) {
72 CATCH_ENFORCE( !isReservedTag(tag),
73 "Tag name: [" << tag << "] is not allowed.\n"
74 << "Tag names starting with non alphanumeric characters are reserved\n"
75 << _lineInfo );

83 constexpr StringRef extractFilenamePart(StringRef filename) {
84 size_t lastDot = filename.size();
85 while (lastDot > 0 && filename[lastDot - 1] != '.') {
86 --lastDot;
87 }
88 // In theory we could have filename without any extension in it
89 if ( lastDot == 0 ) { return StringRef(); }
90
91 --lastDot;
92 size_t nameStart = lastDot;
93 while (nameStart > 0 && filename[nameStart - 1] != '/' && filename[nameStart - 1] != '\\') {
94 --nameStart;
95 }
96
97 return filename.substr(nameStart, lastDot - nameStart);
98 }

100 // Returns the upper bound on size of extra tags ([#file]+[.])
101 constexpr size_t sizeOfExtraTags(StringRef filepath) {
102 // [.] is 3, [#] is another 3
103 const size_t extras = 3 + 3;
104 return extractFilenamePart(filepath).size() + extras;

124 TestCaseInfo::TestCaseInfo(StringRef _className,
125 NameAndTags const& _nameAndTags,
126 SourceLineInfo const& _lineInfo):
127 name( _nameAndTags.name.empty() ? makeDefaultName() : _nameAndTags.name ),
128 className( _className ),
129 lineInfo( _lineInfo )
130 {
131 StringRef originalTags = _nameAndTags.tags;
132 // We need to reserve enough space to store all of the tags
133 // (including optional hidden tag and filename tag)
134 auto requiredSize = originalTags.size() + sizeOfExtraTags(_lineInfo.file);
135 backingTags.reserve(requiredSize);
136
137 // We cannot copy the tags directly, as we need to normalize
138 // some tags, so that [.foo] is copied as [.][foo].
139 size_t tagStart = 0;
140 size_t tagEnd = 0;
141 bool inTag = false;
142 for (size_t idx = 0; idx < originalTags.size(); ++idx) {
143 auto c = originalTags[idx];
144 if (c == '[') {
145 CATCH_ENFORCE(
146 !inTag,
147 "Found '[' inside a tag while registering test case '"
148 << _nameAndTags.name << "' at " << _lineInfo );
149
150 inTag = true;
151 tagStart = idx;
152 }
153 if (c == ']') {
154 CATCH_ENFORCE(
155 inTag,
156 "Found unmatched ']' while registering test case '"
157 << _nameAndTags.name << "' at " << _lineInfo );
158
159 inTag = false;
160 tagEnd = idx;
161 assert(tagStart < tagEnd);
162
163 // We need to check the tag for special meanings, copy
164 // it over to backing storage and actually reference the
165 // backing storage in the saved tags
166 StringRef tagStr = originalTags.substr(tagStart+1, tagEnd - tagStart - 1);
167 CATCH_ENFORCE( !tagStr.empty(),
168 "Found an empty tag while registering test case '"
169 << _nameAndTags.name << "' at "
170 << _lineInfo );
171
172 enforceNotReservedTag(tagStr, lineInfo);
173 properties |= parseSpecialTag(tagStr);
174 // When copying a tag to the backing storage, we need to
175 // check if it is a merged hide tag, such as [.foo], and
176 // if it is, we need to handle it as if it was [foo].
177 if (tagStr.size() > 1 && tagStr[0] == '.') {
178 tagStr = tagStr.substr(1, tagStr.size() - 1);
179 }
180 // We skip over dealing with the [.] tag, as we will add
181 // it later unconditionally and then sort and unique all
182 // the tags.
183 internalAppendTag(tagStr);
184 }
185 }
186 CATCH_ENFORCE( !inTag,
187 "Found an unclosed tag while registering test case '"
188 << _nameAndTags.name << "' at " << _lineInfo );
189
190
191 // Add [.] if relevant
192 if (isHidden()) {
193 internalAppendTag("."_sr);
194 }
195
196 // Sort and prepare tags
197 std::sort(begin(tags), end(tags));
198 tags.erase(std::unique(begin(tags), end(tags)),
199 end(tags));
200 }

221 std::string TestCaseInfo::tagsAsString() const {
222 std::string ret;
223 // '[' and ']' per tag
224 std::size_t full_size = 2 * tags.size();
225 for (const auto& tag : tags) {
226 full_size += tag.original.size();
227 }
228 ret.reserve(full_size);
229 for (const auto& tag : tags) {
230 ret.push_back('[');
231 ret += tag.original;
232 ret.push_back(']');
233 }
234
235 return ret;
236 }

238 void TestCaseInfo::internalAppendTag(StringRef tagStr) {
239 backingTags += '[';
240 const auto backingStart = backingTags.size();
241 backingTags += tagStr;
242 const auto backingEnd = backingTags.size();
243 backingTags += ']';
244 tags.emplace_back(StringRef(backingTags.c_str() + backingStart, backingEnd - backingStart));
245 }

247 bool operator<( TestCaseInfo const& lhs, TestCaseInfo const& rhs ) {
248 // We want to avoid redoing the string comparisons multiple times,
249 // so we store the result of a three-way comparison before using
250 // it in the actual comparison logic.
251 const auto cmpName = lhs.name.compare( rhs.name );
252 if ( cmpName != 0 ) {
253 return cmpName < 0;
254 }
255 const auto cmpClassName = lhs.className.compare( rhs.className );
256 if ( cmpClassName != 0 ) {
257 return cmpClassName < 0;
258 }
259 return lhs.tags < rhs.tags;
260 }
