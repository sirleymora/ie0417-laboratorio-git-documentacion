34 template <typename URng, typename Estimator>
35 static sample
36 resample( URng& rng,
37 unsigned int resamples,
38 double const* first,
39 double const* last,
40 Estimator& estimator ) {
41 auto n = static_cast<size_t>( last - first );
42 Catch::uniform_integer_distribution<size_t> dist( 0, n - 1 );
43
44 sample out;
45 out.reserve( resamples );
46 std::vector<double> resampled;
47 resampled.reserve( n );
48 for ( size_t i = 0; i < resamples; ++i ) {
49 resampled.clear();
50 for ( size_t s = 0; s < n; ++s ) {
51 resampled.push_back( first[dist( rng )] );
52 }
53 const auto estimate =
54 estimator( resampled.data(), resampled.data() + resampled.size() );
55 out.push_back( estimate );
56 }
57 std::sort( out.begin(), out.end() );
58 return out;
59 }

61 static double outlier_variance( Estimate<double> mean,
62 Estimate<double> stddev,
63 int n ) {
64 double sb = stddev.point;
65 double mn = mean.point / n;
66 double mg_min = mn / 2.;
67 double sg = (std::min)( mg_min / 4., sb / std::sqrt( n ) );
68 double sg2 = sg * sg;
69 double sb2 = sb * sb;
70
71 auto c_max = [n, mn, sb2, sg2]( double x ) -> double {
72 double k = mn - x;
73 double d = k * k;
74 double nd = n * d;
75 double k0 = -n * nd;
76 double k1 = sb2 - n * sg2 + nd;
77 double det = k1 * k1 - 4 * sg2 * k0;
78 return static_cast<int>( -2. * k0 /
79 ( k1 + std::sqrt( det ) ) );
80 };
81
82 auto var_out = [n, sb2, sg2]( double c ) {
83 double nc = n - c;
84 return ( nc / n ) * ( sb2 - nc * sg2 );
85 };
86
87 return (std::min)( var_out( 1 ),
88 var_out(
89 (std::min)( c_max( 0. ),
90 c_max( mg_min ) ) ) ) /
91 sb2;
92 }

170 static double
171 standard_deviation( double const* first, double const* last ) {
172 auto m = Catch::Benchmark::Detail::mean( first, last );
173 double variance =
174 std::accumulate( first,
175 last,
176 0.,
177 [m]( double a, double b ) {
178 double diff = b - m;
179 return a + diff * diff;
180 } ) /
181 static_cast<double>( last - first );
182 return std::sqrt( variance );
183 }

185 static sample jackknife( double ( *estimator )( double const*,
186 double const* ),
187 double* first,
188 double* last ) {
189 const auto second = first + 1;
190 sample results;
191 results.reserve( static_cast<size_t>( last - first ) );
192
193 for ( auto it = first; it != last; ++it ) {
194 std::iter_swap( it, first );
195 results.push_back( estimator( second, last ) );
196 }
197
198 return results;
199 }

211 double weighted_average_quantile( int k,
212 int q,
213 double* first,
214 double* last ) {
215 auto count = last - first;
216 double idx = static_cast<double>((count - 1) * k) / static_cast<double>(q);
217 int j = static_cast<int>(idx);
218 double g = idx - j;
219 std::nth_element(first, first + j, last);
220 auto xj = first[j];
221 if ( Catch::Detail::directCompare( g, 0 ) ) {
222 return xj;
223 }
224
225 auto xj1 = *std::min_element(first + (j + 1), last);
226 return xj + g * (xj1 - xj);
227 }

229 OutlierClassification
230 classify_outliers( double const* first, double const* last ) {
231 std::vector<double> copy( first, last );
232
233 auto q1 = weighted_average_quantile( 1, 4, copy.data(), copy.data() + copy.size() );
234 auto q3 = weighted_average_quantile( 3, 4, copy.data(), copy.data() + copy.size() );
235 auto iqr = q3 - q1;
236 auto los = q1 - ( iqr * 3. );
237 auto lom = q1 - ( iqr * 1.5 );
238 auto him = q3 + ( iqr * 1.5 );
239 auto his = q3 + ( iqr * 3. );
240
241 OutlierClassification o;
242 for ( ; first != last; ++first ) {
243 const double t = *first;
244 if ( t < los ) {
245 ++o.low_severe;
246 } else if ( t < lom ) {
247 ++o.low_mild;
248 } else if ( t > his ) {
249 ++o.high_severe;
250 } else if ( t > him ) {
251 ++o.high_mild;
252 }
253 ++o.samples_seen;
254 }
255 return o;
256 }

258 double mean( double const* first, double const* last ) {
259 auto count = last - first;
260 double sum = 0.;
261 while (first != last) {
262 sum += *first;
263 ++first;
264 }
265 return sum / static_cast<double>(count);
266 }

276 double normal_quantile(double p) {
277 static const double ROOT_TWO = std::sqrt(2.0);
278
279 double result = 0.0;
280 assert(p >= 0 && p <= 1);
281 if (p < 0 || p > 1) {
282 return result;
283 }
284
285 result = -erfc_inv(2.0 * p);
286 // result *= normal distribution standard deviation (1.0) * sqrt(2)
287 result *= /*sd * */ ROOT_TWO;
288 // result += normal disttribution mean (0)
289 return result;
290 }

292 Estimate<double>
293 bootstrap( double confidence_level,
294 double* first,
295 double* last,
296 sample const& resample,
297 double ( *estimator )( double const*, double const* ) ) {
298 auto n_samples = last - first;
299
300 double point = estimator( first, last );
301 // Degenerate case with a single sample
302 if ( n_samples == 1 )
303 return { point, point, point, confidence_level };
304
305 sample jack = jackknife( estimator, first, last );
306 double jack_mean =
307 mean( jack.data(), jack.data() + jack.size() );
308 double sum_squares = 0, sum_cubes = 0;
309 for ( double x : jack ) {
310 auto difference = jack_mean - x;
311 auto square = difference * difference;
312 auto cube = square * difference;
313 sum_squares += square;
314 sum_cubes += cube;
315 }
316
317 double accel = sum_cubes / ( 6 * std::pow( sum_squares, 1.5 ) );
318 long n = static_cast<long>( resample.size() );
319 double prob_n = static_cast<double>(
320 std::count_if( resample.begin(),
321 resample.end(),
322 [point]( double x ) { return x < point; } )) /
323 static_cast<double>( n );
324 // degenerate case with uniform samples
325 if ( Catch::Detail::directCompare( prob_n, 0. ) ) {
326 return { point, point, point, confidence_level };
327 }
328
329 double bias = normal_quantile( prob_n );
330 double z1 = normal_quantile( ( 1. - confidence_level ) / 2. );
331
332 auto cumn = [n]( double x ) -> long {
333 return std::lround( normal_cdf( x ) *
334 static_cast<double>( n ) );
335 };
336 auto a = [bias, accel]( double b ) {
337 return bias + b / ( 1. - accel * b );
338 };
339 double b1 = bias + z1;
340 double b2 = bias - z1;
341 double a1 = a( b1 );
342 double a2 = a( b2 );
343 auto lo = static_cast<size_t>( (std::max)( cumn( a1 ), 0l ) );
344 auto hi =
345 static_cast<size_t>( (std::min)( cumn( a2 ), n - 1 ) );
346
347 return { point, resample[lo], resample[hi], confidence_level };
348 }

350 bootstrap_analysis analyse_samples(double confidence_level,
351 unsigned int n_resamples,
352 double* first,
353 double* last) {
354 auto mean = &Detail::mean;
355 auto stddev = &standard_deviation;
356
357#if defined(CATCH_CONFIG_USE_ASYNC)
358 auto Estimate = [=](double(*f)(double const*, double const*)) {
359 std::random_device rd;
360 auto seed = rd();
361 return std::async(std::launch::async, [=] {
362 SimplePcg32 rng( seed );
363 auto resampled = resample(rng, n_resamples, first, last, f);
364 return bootstrap(confidence_level, first, last, resampled, f);
365 });
366 };
367
368 auto mean_future = Estimate(mean);
369 auto stddev_future = Estimate(stddev);
370
371 auto mean_estimate = mean_future.get();
372 auto stddev_estimate = stddev_future.get();
373#else
374 auto Estimate = [=](double(*f)(double const* , double const*)) {
375 std::random_device rd;
376 auto seed = rd();
377 SimplePcg32 rng( seed );
378 auto resampled = resample(rng, n_resamples, first, last, f);
379 return bootstrap(confidence_level, first, last, resampled, f);
380 };
381
382 auto mean_estimate = Estimate(mean);
383 auto stddev_estimate = Estimate(stddev);
384#endif // CATCH_USE_ASYNC
385
386 auto n = static_cast<int>(last - first); // seriously, one can't use integral types without hell in C++
387 double outlier_variance = Detail::outlier_variance(mean_estimate, stddev_estimate, n);
388
389 return { mean_estimate, stddev_estimate, outlier_variance };
390 }
