19 // Extracts the actual name part of an enum instance
20 // In other words, it returns the Blue part of Bikeshed::Colour::Blue
21 StringRef extractInstanceName(StringRef enumInstance) {
22 // Find last occurrence of ":"
23 size_t name_start = enumInstance.size();
24 while (name_start > 0 && enumInstance[name_start - 1] != ':') {
25 --name_start;
26 }
27 return enumInstance.substr(name_start, enumInstance.size() - name_start);
28 }
29 }

31 std::vector<StringRef> parseEnums( StringRef enums ) {
32 auto enumValues = splitStringRef( enums, ',' );
33 std::vector<StringRef> parsed;
34 parsed.reserve( enumValues.size() );
35 for( auto const& enumValue : enumValues ) {
36 parsed.push_back(trim(extractInstanceName(enumValue)));
37 }
38 return parsed;
39 }

41 StringRef EnumInfo::lookup( int64_t value ) const {
42 for( auto const& valueToName : m_values ) {

49 EnumInfo makeEnumInfo( StringRef enumName, StringRef allValueNames, std::vector<int64_t> const& values ) {
50 EnumInfo enumInfo;
51 enumInfo.m_name = enumName;
52 enumInfo.m_values.reserve( values.size() );
53
54 const auto valueNames = Catch::Detail::parseEnums( allValueNames );
55 assert( valueNames.size() == values.size() );
56 for (size_t i = 0; i < values.size(); ++i) {
57 enumInfo.m_values.emplace_back( values[i], valueNames[i] );
58 }
59
60 return enumInfo;
61 }
