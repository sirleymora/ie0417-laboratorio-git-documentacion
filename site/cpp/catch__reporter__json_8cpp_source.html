48 JsonReporter::JsonReporter( ReporterConfig&& config ):
49 StreamingReporterBase{ CATCH_MOVE( config ) } {
50
51 m_preferences.shouldRedirectStdOut = true;
52 // TBD: Do we want to report all assertions? XML reporter does
53 // not, but for machine-parseable reporters I think the answer
54 // should be yes.
55 m_preferences.shouldReportAllAssertions = true;
56 // We only handle assertions when they end
57 m_preferences.shouldReportAllAssertionStarts = false;
58
59 m_objectWriters.emplace( m_stream );
60 m_writers.emplace( Writer::Object );
61 auto& writer = m_objectWriters.top();
62
63 writer.write( "version"_sr ).write( kJsonOutputVersion );
64
65 {
66 auto metadata_writer = writer.write( "metadata"_sr ).writeObject();
67 metadata_writer.write( "name"_sr ).write( m_config->name() );
68 metadata_writer.write( "rng-seed"_sr ).write( m_config->rngSeed() );
69 metadata_writer.write( "catch2-version"_sr )
70 .write( libraryVersion() );
71 if ( m_config->testSpec().hasFilters() ) {
72 metadata_writer.write( "filters"_sr )
73 .write( m_config->testSpec() );
74 }
75 }

174 void JsonReporter::testCaseStarting( TestCaseInfo const& tcInfo ) {
175 StreamingReporterBase::testCaseStarting( tcInfo );
176
177 assert( isInside( Writer::Array ) &&
178 "We should be in the 'test-cases' array" );
179 startObject();
180 // "test-info" prelude
181 {
182 auto testInfo =
183 m_objectWriters.top().write( "test-info"_sr ).writeObject();
184 // TODO: handle testName vs className!!
185 testInfo.write( "name"_sr ).write( tcInfo.name );
186 writeSourceInfo(testInfo, tcInfo.lineInfo);
187 writeTags( testInfo.write( "tags"_sr ).writeArray(), tcInfo.tags );
188 writeProperties( testInfo.write( "properties"_sr ).writeArray(),
189 tcInfo );
190 }
191
192
193 // Start the array for individual test runs (testCasePartial pairs)
194 startArray( "runs"_sr );
195 }

339 void JsonReporter::listTests( std::vector<TestCaseHandle> const& tests ) {
340 startListing();
341
342 auto writer = m_objectWriters.top().write( "tests"_sr ).writeArray();
343
344 for ( auto const& test : tests ) {
345 auto desc_writer = writer.writeObject();
346 auto const& info = test.getTestCaseInfo();
347
348 desc_writer.write( "name"_sr ).write( info.name );
349 if (!info.className.empty()) {
350 desc_writer.write( "class-name"_sr ).write( info.className );
351 }
352 if ( m_verbosity >= Verbosity::Normal ) {
353 auto tag_writer = desc_writer.write( "tags"_sr ).writeArray();
354 for ( auto const& tag : info.tags ) {
355 tag_writer.write( tag.original );
356 }
357 }
358 if ( m_verbosity >= Verbosity::High) {
359 writeSourceInfo( desc_writer, info.lineInfo );
360 }
361 }
362 }
363 void JsonReporter::listTags( std::vector<TagInfo> const& tags ) {

373 aliases_writer.write( alias );
