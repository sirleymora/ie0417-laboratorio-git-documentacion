69 void
70 CumulativeReporterBase::sectionStarting( SectionInfo const& sectionInfo ) {
71 // We need a copy, because SectionStats expect to take ownership
72 SectionStats incompleteStats( SectionInfo(sectionInfo), Counts(), 0, false );
73 SectionNode* node;
74 if ( m_sectionStack.empty() ) {
75 if ( !m_rootSection ) {
76 m_rootSection =
77 Detail::make_unique<SectionNode>( incompleteStats );
78 }
79 node = m_rootSection.get();
80 } else {
81 SectionNode& parentNode = *m_sectionStack.back();
82 auto it = std::find_if( parentNode.childSections.begin(),
83 parentNode.childSections.end(),
84 BySectionInfo( sectionInfo ) );
85 if ( it == parentNode.childSections.end() ) {
86 auto newNode =
87 Detail::make_unique<SectionNode>( incompleteStats );
88 node = newNode.get();
89 parentNode.childSections.push_back( CATCH_MOVE( newNode ) );
90 } else {
91 node = it->get();
92 }
93 }
94
95 m_deepestSection = node;
96 m_sectionStack.push_back( node );
97 }

99 void CumulativeReporterBase::assertionEnded(
100 AssertionStats const& assertionStats ) {
101 assert( !m_sectionStack.empty() );
102 // AssertionResult holds a pointer to a temporary DecomposedExpression,
103 // which getExpandedExpression() calls to build the expression string.
104 // Our section stack copy of the assertionResult will likely outlive the
105 // temporary, so it must be expanded or discarded now to avoid calling
106 // a destroyed object later.
107 if ( m_shouldStoreFailedAssertions &&
108 !assertionStats.assertionResult.isOk() ) {
109 static_cast<void>(
110 assertionStats.assertionResult.getExpandedExpression() );
111 }
112 if ( m_shouldStoreSuccesfulAssertions &&
113 assertionStats.assertionResult.isOk() ) {
114 static_cast<void>(
115 assertionStats.assertionResult.getExpandedExpression() );
116 }
117 SectionNode& sectionNode = *m_sectionStack.back();
118 sectionNode.assertionsAndBenchmarks.emplace_back( assertionStats );
119 }

121 void CumulativeReporterBase::sectionEnded( SectionStats const& sectionStats ) {
122 assert( !m_sectionStack.empty() );
123 SectionNode& node = *m_sectionStack.back();

128 void CumulativeReporterBase::testCaseEnded(
129 TestCaseStats const& testCaseStats ) {
130 auto node = Detail::make_unique<TestCaseNode>( testCaseStats );
131 assert( m_sectionStack.size() == 0 );
132 node->children.push_back( CATCH_MOVE(m_rootSection) );
133 m_testCases.push_back( CATCH_MOVE(node) );
134
135 assert( m_deepestSection );
136 m_deepestSection->stdOut = testCaseStats.stdOut;
137 m_deepestSection->stdErr = testCaseStats.stdErr;
138 }
