20 // This class encapsulates the idea of a pool of ostringstreams that can be reused.
21 struct StringStreams {
22 std::vector<Detail::unique_ptr<std::ostringstream>> m_streams;
23 std::vector<std::size_t> m_unused;
24 std::ostringstream m_referenceStream; // Used for copy state/ flags from
25 Detail::Mutex m_mutex;
26
27 auto add() -> std::pair<std::size_t, std::ostringstream*> {
28 Detail::LockGuard _( m_mutex );
29 if( m_unused.empty() ) {
30 m_streams.push_back( Detail::make_unique<std::ostringstream>() );
31 return { m_streams.size()-1, m_streams.back().get() };
32 }
33 else {
34 auto index = m_unused.back();
35 m_unused.pop_back();
36 return { index, m_streams[index].get() };
37 }
38 }
39
40 void release( std::size_t index, std::ostream* originalPtr ) {
41 assert( originalPtr );
42 originalPtr->copyfmt( m_referenceStream ); // Restore initial flags and other state
43
44 Detail::LockGuard _( m_mutex );
45 assert( originalPtr == m_streams[index].get() && "Mismatch between release index and stream ptr" );
46 m_unused.push_back( index );
47 }
48 };
